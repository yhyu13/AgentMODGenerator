"""API request/response schemas."""
from datetime import datetime
from typing import Any, Literal

from pydantic import BaseModel, Field, field_validator

# v52 Red: constant mirrors ``generators.packager._MAX_LOCALES_PER_PACK``
# (8). We import the value via a deferred import inside the validator
# helper (below) to avoid the AGENTS.md "test isolation: no top-level
# ``app.config`` import" rule — schemas.py is imported transitively
# by test files that don't want config side-effects. The constant
# itself is defined inline so the schema layer doesn't take a hard
# import dependency on ``generators.packager`` either; the validator
# re-fetches the real value at validation time and this fallback is
# only used if the import fails.
_API_MAX_LOCALES_PER_PACK: int = 8


def _validate_locales_field(value: list[str]) -> list[str]:
    """Validate the ``locales`` list on ``GenerateRequest`` / ``BatchGenerateRequest``.

    v52 Red: thin wrapper around
    :func:`generators.packager._validate_locale_code` that enforces
    three rules at the API boundary:

    1. No more than ``_MAX_LOCALES_PER_PACK`` entries (8). Anything
       more is rejected with a 422 from Pydantic before the request
       ever hits the orchestrator.
    2. No duplicates — callers passing ``["fr", "fr"]`` get a single
       ``"fr"`` back. This is a usability win (operators often copy
       locale lists and forget to dedup) and keeps the per-request
       zip inventory deterministic.
    3. Each entry must satisfy
       :func:`generators.packager._validate_locale_code` (BCP-47
       subset, no path traversal, no null bytes).

    The packager helper is imported inside the function body (not at
    module top-level) to keep this module import-safe for tests that
    don't want the ``generators.packager`` import chain. The autouse
    ``_isolate_test_env`` fixture in ``tests/conftest.py`` doesn't
    affect ``generators.packager``, but deferring the import keeps the
    dependency optional and prevents a future refactor from breaking
    every test that imports ``app.api.schemas``.

    The function returns the validated (and deduplicated) list so the
    field value is what gets serialised back through ``GenerateResponse``,
    giving the caller an immediate confirmation of exactly what the
    server accepted.
    """
    # Deferred import — see docstring. We also pull the cap value from
    # the packager module so the schema and the wrapper stay in sync;
    # the local ``_API_MAX_LOCALES_PER_PACK`` constant above is the
    # last-resort fallback if that import fails.
    try:
        from generators.packager import _MAX_LOCALES_PER_PACK, _validate_locale_code
    except ImportError:  # pragma: no cover — defensive fallback only
        max_locales = _API_MAX_LOCALES_PER_PACK

        def _validate_locale_code(lang: str) -> str:  # type: ignore[misc]
            raise ValueError(f"locale validation unavailable (got {lang!r})")

    else:
        max_locales = _MAX_LOCALES_PER_PACK

    if len(value) > max_locales:
        raise ValueError(
            f"too many locales: {len(value)} > {max_locales} "
            f"(generators.packager._MAX_LOCALES_PER_PACK)"
        )
    # Preserve order while deduplicating so callers passing
    # ``["fr", "de", "fr"]`` get ``["fr", "de"]`` back (stable).
    seen: set[str] = set()
    deduped: list[str] = []
    for raw in value:
        if not isinstance(raw, str):
            raise ValueError(
                f"locale must be a string (got {type(raw).__name__})"
            )
        normalised = _validate_locale_code(raw)
        if normalised not in seen:
            seen.add(normalised)
            deduped.append(normalised)
    return deduped


def _sanitize_user_id(value: str) -> str:
    """Normalize user_id: strip whitespace and reject dangerous chars.

    Strips surrounding whitespace (common when users paste IDs from chat
    clients) and rejects newlines / control characters that could break
    log lines or downstream SQL LIKE queries.
    """
    cleaned = value.strip()
    if not cleaned:
        return cleaned
    for ch in ("\n", "\r", "\t", "\x00"):
        if ch in cleaned:
            raise ValueError(
                f"user_id must not contain control characters (got {value!r})"
            )
    return cleaned


def _sanitize_prompt(value: str) -> str:
    """Normalize prompt: strip whitespace, collapse internal whitespace runs.

    We don't truncate here (max_length handles that) but we trim
    surrounding whitespace and reject null bytes. Internal whitespace is
    preserved verbatim — collapsing could change the meaning of a
    prompt that mentions code or file paths.
    """
    cleaned = value.strip()
    if "\x00" in cleaned:
        raise ValueError("prompt must not contain null bytes")
    return cleaned


class GenerateRequest(BaseModel):
    user_id: str = Field(
        min_length=1,
        max_length=128,
        description=(
            "Stable identifier for the requesting user. Sanitized at the schema "
            "boundary (stripped + control chars rejected) so log lines and "
            "downstream SQL LIKE queries can't be broken."
        ),
    )
    prompt: str = Field(
        min_length=1,
        max_length=10000,
        description=(
            "Natural-language prompt describing the mod to generate. Stripped of "
            "surrounding whitespace; null bytes rejected. 10000-char cap matches "
            "the mod generator's prompt budget."
        ),
    )
    phase: str | None = Field(default=None, description="Optional phase override (e.g. shop_channel, weather_event).")
    tags: list[str] = Field(default_factory=list, description="Optional tags for the mod request.")
    prompt_safety: bool = Field(
        default=True,
        description=(
            "When True (default), the request prompt is run through "
            "app.prompt_sanitizer.sanitize_prompt before being logged, "
            "persisted to mod_requests, or forwarded to the LLM. "
            "Set to False to opt out of sanitization for trusted "
            "internal callers (admin tooling, integration tests, "
            "automation) where redaction would interfere with the "
            "intended payload. The endpoint itself does not consume "
            "this flag yet — it is a forward-compatible knob that "
            "downstream wiring will read once sanitize_prompt is on "
            "the request hot path."
        ),
    )
    with_rewards: bool = Field(
        default=False,
        description=(
            "When True (default False), the daily_newsletter pipeline emits a "
            "non-empty assets/newsletter/rewards.json with deterministic per-"
            "weekday item attachments (parsnip on Monday, salmonberry on "
            "Thursday, etc.). When False (the default), the rewards generator "
            "still runs but emits the empty placeholder so the orchestration "
            "slot exists. Only honoured when phase=daily_newsletter; ignored "
            "for all other phases."
        ),
    )
    locales: list[str] = Field(
        default_factory=list,
        max_length=_API_MAX_LOCALES_PER_PACK,
        description=(
            "Optional BCP-47 locale codes to emit as per-language "
            "``i18n/<lang>.json`` files inside the packaged mod zip. "
            "Empty list (default) keeps the v51 zero-cost path: the "
            "packager ships only the auto-injected ``i18n/default.json`` "
            "skeleton and no per-language files. A non-empty list "
            "activates :func:`generators.packager.package_with_locales` "
            "in the orchestrator's ``node_package``. The list is "
            "validated at the schema boundary — duplicates are removed, "
            "and any entry that fails "
            ":func:`generators.packager._validate_locale_code` raises "
            "a 422 before the request reaches the pipeline. The "
            "accepted (deduplicated) list is echoed back in "
            ":attr:`GenerateResponse.locales` so callers can verify "
            "exactly what the server accepted. Cap matches "
            "``generators.packager._MAX_LOCALES_PER_PACK`` (8)."
        ),
    )

    @field_validator("user_id")
    @classmethod
    def _validate_user_id(cls, v: str) -> str:
        return _sanitize_user_id(v)

    @field_validator("prompt")
    @classmethod
    def _validate_prompt(cls, v: str) -> str:
        return _sanitize_prompt(v)

    @field_validator("locales")
    @classmethod
    def _validate_locales(cls, v: list[str]) -> list[str]:
        # Delegates to the module-level helper so BatchGenerateRequest
        # can reuse the exact same rule set without duplicating logic.
        return _validate_locales_field(v)


class BatchGenerateRequest(BaseModel):
    user_id: str = Field(
        min_length=1,
        max_length=128,
        description=(
            "Stable identifier for the requesting user. Shared across every "
            "prompt in the batch (each prompt becomes its own request_id)."
        ),
    )
    prompts: list[str] = Field(
        min_length=1,
        max_length=10,
        description=(
            "1-10 prompts to run in parallel. Each prompt gets its own request_id "
            "but they share the same user_id, phase, and tags."
        ),
    )
    phase: str | None = Field(default=None, description="Optional phase override for all items.")
    tags: list[str] = Field(default_factory=list, description="Optional tags for the batch request.")
    prompt_safety: bool = Field(
        default=True,
        description=(
            "When True (default), every prompt in ``prompts`` is run through "
            "app.prompt_sanitizer.sanitize_prompt before being logged, "
            "persisted to mod_requests, or forwarded to the LLM. "
            "Set to False to opt out of sanitization for trusted "
            "internal callers (admin tooling, integration tests, "
            "automation) where redaction would interfere with the "
            "intended payload. The endpoint reads this flag once and "
            "applies it to every item in the batch, so the cost is "
            "linear in the batch size and the flag cannot be "
            "selectively toggled per-item."
        ),
    )
    locales: list[str] = Field(
        default_factory=list,
        max_length=_API_MAX_LOCALES_PER_PACK,
        description=(
            "Optional BCP-47 locale codes broadcast to every item in "
            "the batch (mirrors the ``locales`` field on "
            ":class:`GenerateRequest`). The validation, dedup, and "
            "cap rules are identical — see ``GenerateRequest.locales`` "
            "for the contract. The list is read once by the batch "
            "handler and forwarded to each per-item "
            "``run_pipeline_background`` call so every prompt in the "
            "batch produces per-locale ``i18n/<lang>.json`` files."
        ),
    )
    with_rewards: bool = Field(
        default=False,
        description=(
            "Forwarded to every item in the batch as the per-item "
            "``with_rewards`` flag. Mirrors the field on "
            ":class:`GenerateRequest` exactly — see that field's "
            "description for the daily_newsletter semantics. Read "
            "once by the batch handler and broadcast to each "
            "per-item ``run_pipeline_background`` call; cannot be "
            "toggled per-item within a single batch."
        ),
    )

    @field_validator("user_id")
    @classmethod
    def _validate_user_id(cls, v: str) -> str:
        return _sanitize_user_id(v)

    @field_validator("prompts")
    @classmethod
    def _validate_prompts(cls, v: list[str]) -> list[str]:
        return [_sanitize_prompt(p) for p in v]  # type: ignore[arg-type]  # pydantic ensures list[str] at runtime

    @field_validator("locales")
    @classmethod
    def _validate_locales(cls, v: list[str]) -> list[str]:
        # Same delegation as GenerateRequest — keeps the rule set in
        # one place so future changes (e.g. raising the cap) only need
        # to update the helper.
        return _validate_locales_field(v)

class ModSummaryResponse(BaseModel):
    """A human-readable text summary of a mod request.

    Aggregates the request's status, feature name (from manifest),
    file count, generator list, and T1/T2 quality-gate outcomes into a
    single short text block. Designed for `/v1/mods/{id}/summary` and
    for `!summary` style commands in chat bots.
    """
    request_id: str = Field(description="Server-generated id; matches the create-request id.")
    status: str = Field(description="Current pipeline status (matches ModStatusResponse.status).")
    feature_name: str | None = Field(
        default=None,
        description="Human-readable feature name read from the packaged manifest, if available.",
    )
    mod_id: str | None = Field(
        default=None,
        description="Manifest UniqueID for the mod; matches the public mod folder id.",
    )
    file_count: int = Field(default=0, ge=0, description="Number of generated files.")
    generator_count: int = Field(default=0, ge=0, description="Number of generators that ran.")
    generators: list[str] = Field(
        default_factory=list,
        description="Names of generators that produced output, in execution order.",
    )
    t1_status: str = Field(
        default="unknown",
        description="T1 gate verdict ('passed'/'failed'/'unknown' if the gate never ran).",
    )
    t1_error_count: int = Field(default=0, ge=0, description="Number of distinct T1 errors reported.")
    t2_status: str = Field(
        default="unknown",
        description="T2 gate verdict ('passed'/'failed'/'skipped'/'unknown').",
    )
    t2_score: int | None = Field(default=None, description="Average T2 score (0-10); None when judges skipped.")
    t2_max_score: int | None = Field(default=None, description="Maximum possible T2 score (always 10).")
    t2_passed: bool | None = Field(default=None, description="True if T2 gate passed; None when skipped.")
    cancellation_reason: str | None = Field(
        default=None,
        description=(
            "Why a cancelled request was cancelled. Mirrors the field on "
            "ModStatusResponse — same semantics, surfaced here so the "
            "summary text can mention it without a second round-trip."
        ),
    )
    created_at: str | None = Field(
        default=None,
        description="UTC timestamp the request was created (ISO-8601 string in JSON).",
    )
    summary: str = Field(description="Human-readable text summary.")

class TimelineStage(BaseModel):
    """A single stage of the mod generation pipeline in execution order.

    Returned by ``GET /v1/mods/{id}/timeline``. Each stage carries a
    short id (matching the orchestrator's internal ``status`` names
    like ``routing``/``generating``/``validating``/``reviewing``/
    ``packaging``), a human-readable ``label``, a ``reached`` flag
    (``True`` once the pipeline has moved past this stage, ``False``
    while the request is still earlier), and an ISO-8601 ``at``
    timestamp representing when the stage was entered (or
    ``None`` when not yet reached).

    ``at`` is **derived** from existing Redis state (``created_at``
    plus ``duration_seconds`` plus the live ``status`` field) so the
    endpoint does not require any new fields to be written by the
    pipeline. As a result the timestamps are best-effort
    approximations of when each stage *would have* run, not exact
    measurements — callers that need exact per-stage timing should
    add explicit stage logging to the orchestrator.
    """
    stage: str = Field(
        description="Short stage id (e.g. 'routing', 'generating', 'validating')."
    )
    label: str = Field(
        description="Human-readable label for the stage (e.g. 'Routing', 'Generating')."
    )
    reached: bool = Field(
        description="True once the pipeline has moved past this stage."
    )
    current: bool = Field(
        description="True if the pipeline is currently in this stage."
    )
    at: str | None = Field(
        default=None,
        description=(
            "ISO-8601 UTC timestamp at which the stage was entered. "
            "Approximated from created_at + duration_seconds for "
            "completed stages; None while a stage has not yet been "
            "reached."
        ),
    )


class ModTimelineResponse(BaseModel):
    """Response for ``GET /v1/mods/{id}/timeline``.

    Surfaces the per-stage execution order, which stage is currently
    active, and approximate timestamps so operators and chat bots can
    show the user "where it is right now" without re-parsing the full
    status payload. The endpoint is read-only and has no side
    effects — it only reads Redis pipeline state plus the DB
    fallback path used by ``GET /v1/mods/{id}``.

    All ``*_at`` fields are derived from existing state and are
    best-effort approximations, not exact measurements. The
    ``stages`` list is always returned in pipeline execution order
    (routing -> generating -> validating -> reviewing -> packaging ->
    done) so callers can render a fixed-length progress bar without
    re-sorting.
    """
    request_id: str = Field(description="Server-generated id; matches the {id} path parameter.")
    status: str = Field(
        description=(
            "Current pipeline status. One of the standard status "
            "ids (pending, routing, generating, t1_gating, t2_gating, "
            "packaging, done, failed, cancelled)."
        ),
    )
    started_at: str | None = Field(
        default=None,
        description=(
            "ISO-8601 UTC timestamp at which the request was "
            "received. Mirrors the ``created_at`` field on "
            "ModStatusResponse."
        ),
    )
    completed_at: str | None = Field(
        default=None,
        description=(
            "ISO-8601 UTC timestamp at which the request reached a "
            "terminal status (done/failed/cancelled). None while the "
            "request is still in flight."
        ),
    )
    progress_percent: int = Field(
        ge=0, le=100,
        description="Pipeline progress percentage, derived from status.",
    )
    current_stage: str = Field(
        description=(
            "Id of the currently-active stage (matches a ``stage`` "
            "field on one of the entries in ``stages``)."
        ),
    )
    current_stage_label: str = Field(
        description="Human-readable label for the current stage."
    )
    stages: list[TimelineStage] = Field(
        description=(
            "Pipeline stages in execution order. Each entry has a "
            "``reached`` and ``current`` flag plus an optional ``at`` "
            "timestamp."
        ),
    )

    generator_count: int = Field(ge=0, description="Number of generators registered for this phase.")

def _truncate_prompt(value: str | None) -> str | None:
    """Truncate a prompt to 200 characters at the schema boundary.

    The route layer passes the raw prompt through unchanged; this
    helper enforces the 200-char cap that
    :attr:`ModListItem.prompt`'s ``max_length=200`` documents so the
    serialization can't blow up on a multi-KB prompt. Used as a
    ``field_validator`` target so the cap is enforced whether the
    field is populated directly by a route or by
    ``model_validate(row)`` in a test.
    """
    if value is None:
        return None
    return value[:200]

class FlagHistoryEntry(BaseModel):
    """One row in the in-memory feature-flag audit log.

    Returned by ``GET /v1/feature_flags/history`` for every recorded
    ``feature_flag.changed`` event. The fields mirror the structlog
    event emitted by :func:`orchestrator.feature_flags.set_flag` so
    the API and the log shipper cannot drift: any change to the log
    event shape must also update this model, and vice versa.

    The audit log is process-local and resets on every process
    restart — it is intentionally NOT persisted to Redis, the DB,
    or any external store. That is the right trade-off for a
    P5-staged rollout: the recent operator activity is what
    matters (last few hours, last few days), and persisting
    everything would conflate the runtime audit trail with the
    deployment-change history that already lives in git + a future
    change-log endpoint. If long-term persistence becomes
    necessary, the cleanest path is to subscribe a sink to the
    ``feature_flag.changed`` log event and let that sink write to
    whatever store the operator chooses — the in-memory log stays
    fast and bounded, and the durable copy lives wherever the
    operator wants it.

    ``no_op=True`` marks a write that did not mutate the registry
    (previous_value == new_value). These rows are kept on purpose
    so the history captures every operator call (including
    duplicate toggles from a flaky dashboard), but downstream
    consumers can filter on the field to isolate real state
    changes.
    """

    flag_name: str = Field(
        description=(
            "Stable snake_case flag identifier (e.g. "
            "'use_s3_presigned', 't2_three_judge_panel', "
            "'discord_completion_webhook'). Matches the keys of "
            "orchestrator.feature_flags._FLAGS."
        ),
    )
    previous_value: bool = Field(
        description=(
            "The flag's value immediately before the change. Equals "
            "``new_value`` when ``no_op`` is True."
        ),
    )
    new_value: bool = Field(
        description=(
            "The flag's value after the change. Equals "
            "``previous_value`` when ``no_op`` is True."
        ),
    )
    changed_at: str = Field(
        description=(
            "Server-side timestamp at which the change was recorded. "
            "ISO-8601 in UTC with a 'Z' suffix (e.g. "
            "'2026-06-20T15:42:11.123456Z'). Microsecond precision "
            "so rapid consecutive toggles on the same flag produce "
            "distinct, sortable rows."
        ),
    )
    no_op: bool = Field(
        description=(
            "True if the call did not mutate the registry "
            "(previous_value == new_value). Recorded for "
            "audit-trail completeness so the history captures "
            "every operator call, not just the ones that "
            "changed state."
        ),
    )


class FlagHistoryResponse(BaseModel):
    """Response for ``GET /v1/feature_flags/history`` — audit log page.

    Surfaces the in-memory audit log of ``feature_flag.changed``
    events so operators can answer "who flipped which flag when"
    without scraping structlog output or grepping a log shipper.
    The endpoint is read-only and mirrors exactly what was emitted
    by the registry's :func:`record_flag_change` helper — there is
    no transformation, no filtering on the server side beyond the
    optional ``flag_name`` query parameter, and no enrichment.

    ``entries`` is sorted by ``changed_at`` ascending so callers
    see the change history in chronological order, oldest first.
    The endpoint clamps ``entries`` to the last ``limit`` rows
    (default 100, max 1000) but ``total`` reflects the FULL count
    BEFORE the limit is applied — so a caller can detect that the
    history has wrapped or grown past the page size and request
    the next page by adjusting the ``flag_name`` filter or by
    paginating with a future ``before`` cursor.

    ``total`` is included even when the response is empty (== 0)
    so clients can assert "the log is genuinely empty" vs. "the
    filter matched nothing" without an extra round-trip.
    """

    entries: list[FlagHistoryEntry] = Field(
        description=(
            "Audit-log rows that matched the request, sorted by "
            "changed_at ascending. Empty list if the history is "
            "empty or the flag_name filter matched no rows."
        ),
    )
    total: int = Field(
        ge=0,
        description=(
            "Total number of rows that matched the request, "
            "BEFORE the ``limit`` clamp is applied. Equal to "
            "len(entries) when the matching set was small enough "
            "to fit in one page."
        ),
    )


class FeatureFlagRollbackResponse(BaseModel):
    """Response for ``POST /v1/feature_flags/{name}/rollback``.

    The v18 companion to ``FeatureFlagChangeResponse`` (the toggle
    endpoint's response): surfaces the result of a rollback so the
    operator can confirm what was undone, what the flag is now,
    and which audit-log entry was the source of truth for the
    restored value. Together with the v15 GET (registry snapshot),
    v16 POST (toggle), and v17 GET (audit log) endpoints, this
    closes the operator-dashboard loop on a single flag: read,
    write, audit, undo.

    Field semantics:

    - ``name`` — the flag that was rolled back. Matches the
      ``{name}`` path parameter that was POSTed.
    - ``rolled_back_from`` — the value of the flag immediately
      before the rollback took effect. This is what the
      ``set_flag`` call inside the rollback helper observed as the
      "previous value" at write time, and it is the value the
      flag held at the end of the most recent real mutation. In
      the common case this equals the ``new_value`` of the most
      recent ``feature_flag.changed`` event for this flag.
    - ``rolled_back_to`` — the value of the flag after the
      rollback. Equals the ``previous_value`` of the most recent
      non-no-op entry in the audit log (i.e. the value the flag
      held BEFORE the change that is now being undone). On a
      second consecutive rollback (undoing the undo), this is the
      value the flag had before the first rollback — the flag
      ends up back where it started before either call.
    - ``restored_entry_index`` — the position (0-indexed, in the
      ascending-sorted history) of the audit-log entry whose
      ``previous_value`` was re-applied. Surfaced so dashboards
      can render "rolled back change #N from the audit log" and
      so a caller can cross-reference the rollback to the exact
      row that motivated it. ``-1`` when the helper was unable
      to find a rollbackable entry — kept in-band (rather than

      ``t2_three_judge_panel`` feature flag is off, 3 otherwise)
    - ``panel_passed_count`` — number of judges with score ``>=
      T2_PASS_THRESHOLD`` for this iteration. ``0`` is a valid
      value (every judge voted below threshold)
    """

    iteration: int = Field(
        ge=1,
        description=(
            "1-based iteration index for this T2 run. Matches the "
            "value of ``PipelineState.t2_iterations`` after the "
            "gate ran."
        ),
    )
    score: int | None = Field(
        default=None,
        ge=0,
        le=10,
        description=(
            "Integer panel-average score in [0, 10]. ``None`` when "
            "the gate errored and a numeric score could not be "
            "computed."
        ),
    )
    feedback: str = Field(
        default="",
        description=(
            "Human-readable aggregation of the per-judge verdicts "
            "for this iteration. Empty string when the gate errored."
        ),
    )
    passed: bool = Field(
        description=(
            "Whether the panel passed the T2 pass threshold for "
            "this iteration. ``True`` when the gate errored and the "
            "pipeline fell back to advisory-shipping the output."
        ),
    )
    panel_scores: list[int] = Field(
        default_factory=list,
        description=(
            "Per-judge scores for this iteration. 1 entry when the "
            "``t2_three_judge_panel`` flag is off, 3 entries when "
            "on."
        ),
    )
    panel_passed_count: int = Field(
        ge=0,
        description=(
            "Number of judges whose score was ``>= T2_PASS_THRESHOLD`` "
            "for this iteration. ``0`` is a valid value (every "
            "judge voted below threshold)."
        ),
    )


class T2JudgesResponse(BaseModel):
    """Response envelope for ``GET /v1/mods/{id}/t2_judges``.

    v47 Red (Feature — per-iteration T2 history endpoint). Surfaces
    the full T2 retry history for a request, complementing the
    single-pass ``t2_score`` / ``t2_feedback`` fields on
    :class:`ModStatusResponse`. The endpoint is intentionally
    cache-first (Redis) because per-iteration data is NOT persisted
    to ``mod_outputs`` — only the final score + feedback is. A
    Redis miss therefore surfaces ``iterations=[]`` and
    ``source="none"`` rather than a 404, because the request itself
    still exists in Postgres (just not in the live pipeline cache
    anymore, either because the pipeline never ran T2 or because
    the 24h TTL expired).

    The ``source`` field tells operators exactly where the response
    came from:

    - ``"redis"`` — the pipeline state was in Redis (live or
      recently completed); ``iterations`` reflects whatever was
      found there.
    - ``"db_unavailable"`` — Redis was unreachable, but the request
      was confirmed to exist in ``mod_outputs``. ``iterations`` is
      empty (per-iteration history is Redis-only) but the request
      is not 404.
    - ``"none"`` — neither Redis nor the DB has the request, or
      the request never ran T2. ``iterations`` is empty.
    """

    request_id: str = Field(
        description="The mod request this iteration history is for.",
    )
    iterations: list[T2JudgeIteration] = Field(
        default_factory=list,
        description=(
            "Per-iteration T2 payloads, ordered ascending by "
            "``iteration`` (i.e. earliest retry first). Empty when "
            "the request never ran T2, when the Redis pipeline "
            "state has expired, or when Redis was unreachable "
            "and the DB fallback confirmed existence but has no "
            "per-iteration history."
        ),
    )
    final_score: int | None = Field(
        default=None,
        description=(
            "Echo of the final T2 panel-average score from the "
            "Redis pipeline state. ``None`` when not available "
            "(Redis miss, request never ran T2, or the gate "
            "errored). Matches ``ModStatusResponse.t2_score``."
        ),
    )
    final_passed: bool | None = Field(
        default=None,
        description=(
            "Echo of the final T2 pass/fail verdict from the Redis "
            "pipeline state. ``None`` when not available. Matches "
            "``ModStatusResponse.t2_passed``."
        ),
    )
    t2_available: bool = Field(
        default=False,
        description=(
            "``True`` when the T2 panel produced a real numeric "
            "score at least once. ``False`` when the gate was "
            "skipped (no LLM provider) or errored for every "
            "iteration. Matches ``ModStatusResponse.t2_available``."
        ),
    )
    source: Literal["redis", "db_unavailable", "none"] = Field(
        default="none",
        description=(
            "Where the response payload came from. ``redis`` "
            "(happy path), ``db_unavailable`` (Redis unreachable "
            "but request exists in DB), or ``none`` (request not "
            "found or never ran T2). Defaults to ``\"none\"`` so a "
            "minimal constructor (``T2JudgesResponse(request_id=...)``) "
            "round-trips cleanly."
        ),
    )


# ---------------------------------------------------------------------------
# v55 Red: prompt-keyed estimate schemas for the new ``GET /v1/estimate``
# and ``POST /v1/estimate/batch`` endpoints.
#
# These complement (do not replace) the existing ``GET /v1/estimates``
# (plural, returns the full phase→seconds table) and
# ``GET /v1/estimates/{phase}`` (returns seconds for a single known phase)
# endpoints. The new singular/batch variants are *prompt-keyed* — the
# caller passes a natural-language prompt and gets back the phase that
# the orchestrator would select plus the corresponding seconds estimate.
#
# Two endpoints share one Pydantic request/response family so the JSON
# shapes line up byte-for-byte between the single-prompt and batch
# variants — a UI that renders "this will take ~60s" cards can use the
# same row type for both.
# ---------------------------------------------------------------------------


class PromptEstimateResponse(BaseModel):
    """Response for ``GET /v1/estimate?prompt=...`` — prompt-keyed estimate.

    Lets a UI render "this will take ~60 seconds" before the user
    submits a full generation request. The endpoint composes two
    already-existing helpers:

    * :func:`orchestrator.router.route` resolves the *phase* the
      orchestrator would pick for the prompt (same path the real
      pipeline takes, including the longest-keyword-wins heuristic).
    * :func:`app.estimation.estimate_seconds_for_phase` returns the
      canonical seconds budget for that phase.

    ``seconds`` is always > 0 — the route guarantees it by passing the
    raw prompt through the estimation helper, which falls back to the
    documented ``_DEFAULT_SECONDS`` (90s) when the phase is unknown.
    The ``matched`` flag is ``True`` iff the resolved phase id is in
    the canonical phase table (i.e. the caller can rely on ``seconds``
    being a tuned estimate rather than the fallback default).

    Mirrors the JSON contract of :class:`PhaseEstimateResponse`
    (phase-keyed lookup) and adds an echoed ``prompt`` field so the
    caller can correlate the response with the request without keeping
    a separate id.
    """

    prompt: str = Field(
        description=(
            "Echo of the (trimmed) prompt the server estimated. Useful "
            "for log correlation and for UIs that render multiple "
            "estimate rows in a list without tracking the request "
            "separately."
        ),
    )
    phase: str = Field(
        description=(
            "Phase id the orchestrator would select for this prompt "
            "(e.g. ``shop_channel``, ``weather_event``). Falls back "
            "to the default phase when no keyword matches."
        ),
    )
    seconds: int = Field(
        ge=1,
        description=(
            "Estimated generation time in seconds. Equals "
            "``app.estimation._PHASE_SECONDS[phase]`` when "
            "``matched`` is True; equals ``default_seconds`` otherwise."
        ),
    )
    default_seconds: int = Field(
        ge=1,
        description=(
            "Fallback estimate surfaced so clients can render 'this "
            "phase is unknown — defaulting to 90s' without a second "
            "round-trip. Mirrors ``app.estimation._DEFAULT_SECONDS``."
        ),
    )
    matched: bool = Field(
        description=(
            "True iff the resolved phase is in the canonical phase "
            "table (i.e. ``seconds`` is a tuned estimate rather than "
            "the fallback default)."
        ),
    )
    game: str = Field(
        default="stardew_valley",
        description=(
            "Game pack the resolved phase belongs to. Mirrors the "
            "``game`` field on ``RoutePreviewResponse`` so the two "
            "preview-style endpoints share a vocabulary. Defaults to "
            "``\"stardew_valley\"`` because the router's fallback "
            "pack is stardew_valley."
        ),
    )


class BatchPromptEstimateItem(BaseModel):
    """One row of the ``POST /v1/estimate/batch`` response.

    Mirrors :class:`PromptEstimateResponse` minus the echoed prompt
    (which is already in the request body) — the batch variant keeps
    the per-row payload small so a UI rendering 10 cards pays ~9
    fewer bytes per row.
    """

    phase: str = Field(description="Resolved phase id for this prompt.")
    seconds: int = Field(
        ge=1,
        description="Estimated generation time in seconds for this prompt.",
    )
    default_seconds: int = Field(
        ge=1,
        description="Fallback estimate (mirrors ``app.estimation._DEFAULT_SECONDS``).",
    )
    matched: bool = Field(
        description=(
            "True iff the resolved phase has a tuned estimate "
            "(not a fallback to ``default_seconds``)."
        ),
    )
    game: str = Field(
        default="stardew_valley",
        description="Game pack the resolved phase belongs to.",
    )


class BatchPromptEstimateRequest(BaseModel):
    """Request body for ``POST /v1/estimate/batch``.

    Accepts 1-20 prompts per batch — small enough to fit in a single
    UI frame for a "what can I generate" card grid, large enough to
    amortise the routing cost on the server. Each prompt gets its own
    per-item estimate in the response, in the same order as the
    request.
    """

    prompts: list[str] = Field(
        min_length=1,
        max_length=20,
        description=(
            "1-20 prompts to estimate in one round-trip. Order is "
            "preserved in the response so callers can render the "
            "results as a list of cards without re-sorting. Each "
            "prompt is trimmed at the schema boundary; empty after "
            "trim is rejected with a 422 before the handler runs."
        ),
    )

    @field_validator("prompts")
    @classmethod
    def _validate_prompts(cls, v: list[str]) -> list[str]:
        # Reuse the same trim + null-byte-rejection rule as
        # ``GenerateRequest._validate_prompt`` so the batch endpoint
        # and the single-prompt endpoint have identical input
        # hygiene. Per-item validation surfaces as a 422 for the
        # whole batch — partial successes would force the caller
        # to walk two parallel lists.
        cleaned: list[str] = []
        for raw in v:
            stripped = raw.strip()
            if not stripped:
                raise ValueError(
                    "every prompt must be non-empty after trim"
                )
            if "\x00" in stripped:
                raise ValueError("prompt must not contain null bytes")
            cleaned.append(stripped)
        return cleaned


class BatchPromptEstimateResponse(BaseModel):
    """Response envelope for ``POST /v1/estimate/batch``.

    ``estimates`` is parallel to the request ``prompts`` list — the
    i-th element is the estimate for the i-th prompt. ``count`` is
    ``len(estimates)`` so callers can sanity-check without a second
    ``len()`` call.
    """

    estimates: list[BatchPromptEstimateItem] = Field(
        description=(
            "Per-prompt estimates, in the same order as the request "
            "``prompts`` list. Length == ``count``."
        ),
    )
    count: int = Field(
        ge=0,
        description="Length of the estimates list (== len(estimates)).",
    )
    default_seconds: int = Field(
        ge=1,
        description=(
            "Echo of the fallback estimate (``app.estimation._DEFAULT_SECONDS``) "
            "so a UI can render the fallback value next to each card "
            "without re-asking the server."
        ),
    )


class HealthDependencyStatus(BaseModel):
    """One row of the ``/v1/health/summary`` table.

    v57 Red: intentionally minimal — just the dependency name and
    whether it is currently up. ``latency_ms`` is included on a
    best-effort basis (the underlying probe records it; the summary
    endpoint forwards it so an operator can spot a slow-but-up
    dependency without paging through the full ``/health/deep``
    payload). When the probe failed before it could measure latency,
    ``latency_ms`` is ``None``.
    """

    name: str = Field(
        description=(
            "Short identifier of the dependency (e.g. ``postgres``, "

            "field used by ``/health/deep``."
        ),
    )
    ok: bool = Field(
        description=(
            "``True`` when the most recent probe of this dependency "
            "succeeded; ``False`` on timeout, connection error, or "

    )
    latency_ms: int | None = Field(
        default=None,
        description=(
            "Wall-clock latency of the most recent probe in "

            "latency (e.g. fast-fail in local mode for S3, or the "
            "Discord bot hasn't measured gateway latency yet)."
        ),
    )


class HealthSummaryResponse(BaseModel):
    """Condensed readiness snapshot for K8s-style probes.

    v57 Red: a one-line-per-dependency view of the same probes that
    power ``/health/deep``. The body is intentionally small (no
    nested ``checks`` list, no error strings, no per-probe ``error``
    text) so a Kubernetes ``livenessProbe`` / ``readinessProbe`` can
    parse it cheaply, and so it is a stable shape for downstream
    alerting tools that just want a table of booleans.

    The condensed view loses two things on purpose:

    1. The full error message per probe (use ``/health/deep`` for
       human diagnosis — this endpoint is for machines).
    2. The structured ``error_type`` field (the same — drill into
       ``/health/deep`` if you need to know *why* a dependency is
       down).

    The HTTP status code matches the aggregate: ``200`` when every
    dependency is up, ``503`` when at least one is down. The body
    shape is identical in both cases so a probe that does not care
    about the status code can still parse the table.
    """

    status: str = Field(
        description=(
            "Aggregate status of the service. ``\"ok\"`` when every "
            "dependency probe succeeded; ``\"degraded\"`` when at "
            "least one failed."
        ),
    )
    dependencies: list[HealthDependencyStatus] = Field(
        description=(
            "One row per dependency, in a stable order (postgres, "
            "redis, s3, discord_bot — same order as ``/health/deep``)."
        ),
    )
    checked_at: str = Field(
        description=(
            "UTC timestamp (ISO-8601) at which the summary was "
            "generated. Useful for cache-staleness checks on the "
            "probe side."
        ),
    )
    # v67 Blue: additive mirror of ``/health/deep``'s ``failed_count``.
    # Operators who route alerts on the deep endpoint now get the same
    # number on the K8s-friendly summary endpoint. Always present so the
    # field shape is uniform across both ``200`` and ``503`` responses
    # and a downstream consumer can read it unconditionally.
    failed_count: int = Field(
        default=0,
        description=(
            "Number of dependency probes that failed in the most recent "
            "probe round (0 when every dependency is up). Mirrors the "
            "``failed_count`` field on ``/health/deep`` so operators "
            "can route alerts on a single number without re-parsing "
            "the ``dependencies`` table."
        ),
        ge=0,
    )
