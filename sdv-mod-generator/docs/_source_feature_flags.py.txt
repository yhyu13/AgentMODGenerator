"""Feature flags — runtime kill-switches for staged P5 rollouts."""
from __future__ import annotations

from datetime import datetime, timezone
from typing import Any, Final

import structlog

logger = structlog.get_logger(__name__)

# Canonical registry of feature flags. Keys are stable, snake_case
# identifiers; values are the on/off switch. Defaults are chosen to
# be safe (off) so a freshly-deployed instance never enables a
# half-built feature. To turn a feature on in a given environment,
# override the value via the deployment layer (see AGENTS.md P5
# section) — this module itself reads only the in-process constant
# and intentionally does not import app.config so it stays cheap
# and side-effect free to import.
#
# Adding a new flag:
#   1. Pick a snake_case name that names the capability, not the
#      rollout stage (e.g. ``use_s3_presigned`` not ``p5_rollout_a``).
#   2. Default to ``False`` so a missed override can't enable
#      unreviewed behavior in production.
#   3. Document the flag here with the rollout owner and a one-line
#      summary of what the flag gates.
_FLAGS: Final[dict[str, bool]] = {
    # When True, /v1/mods/download/{id} returns a presigned S3 URL.
    # When False (default), falls back to a local-file URL via the
    # storage abstraction. Owner: platform.
    "use_s3_presigned": False,
    # When True, the T2 judge panel runs three LLM judges in
    # parallel; when False, runs a single judge (cheaper, used in
    # local-dev). Owner: quality.
    "t2_three_judge_panel": True,
    # When True, the Discord webhook is invoked on completion;
    # when False, completion is silent (useful for staging
    # environments that don't want to spam the bot channel).
    # Owner: discord.
    "discord_completion_webhook": True,
}


# Process-local audit log of every ``feature_flag.changed`` event.
# Backed by a plain Python list (NOT a ``deque`` and NOT a
# thread-safe queue): the feature-flags module does not use a
# ``threading.Lock`` today, and the deny-by-default + small-flag-set
# design means the list does not grow unbounded in practice (each
# change is one append, and there are only three flags). The known
# follow-up is to add a bounded ring buffer (e.g. ``collections.deque``
# with ``maxlen=10_000``) once any deployment starts flipping flags
# faster than humans can read the audit log — at which point the
# endpoint should also surface a ``dropped`` count so operators can
# detect that the buffer wrapped.
#
# Each entry is a flat ``dict[str, Any]`` rather than a dataclass
# so the JSON shape of the eventual ``FlagHistoryEntry`` Pydantic
# model matches the storage shape exactly (one fewer mapping step
# in the API layer).
_HISTORY: list[dict[str, Any]] = []


# Set of flags that have been "pinned" by an operator. A pinned flag
# cannot be mutated by :func:`set_flag` (or by :func:`rollback_flag`,
# which routes through ``set_flag``). Pinning is the operator-facing
# version of "do not change this" — useful when staging a rollout and
# wanting to mark a feature as locked without deleting it from the
# registry. Pin state is process-local (mirrors ``_FLAGS`` and
# ``_HISTORY``) and resets on every process restart.
#
# The set is plain ``set[str]`` rather than ``frozenset`` because the
# register/unregister operations are mutations, and the operation set
# is small (three flags today, likely never more than a handful), so
# the thread-safety overhead of ``set`` vs ``frozenset`` is not worth
# the copy-on-mutate cost. Tests that mutate this state should reset
# it via :func:`clear_pinned_flags` in a ``finally`` block to keep
# the test suite hermetic.
_PINNED_FLAGS: set[str] = set()


def is_enabled(name: str) -> bool:
    """Return whether the named feature flag is on.

    Unknown flags default to ``False`` (deny-by-default) and are
    logged at warning level so a typo or a stale caller is visible
    in operator dashboards without raising. This is intentional:
    a missing flag should fail closed, not crash the pipeline.
    """
    if name in _FLAGS:
        return _FLAGS[name]
    logger.warning(
        "feature_flag.unknown",
        flag_name=name,
        known_flags=sorted(_FLAGS.keys()),
    )
    return False


def known_flags() -> tuple[str, ...]:
    """Return the sorted tuple of registered flag names.

    Exposed for the operator endpoint and for tests that want to
    assert the registry shape without re-parsing this module. The
    tuple is sorted so callers can rely on deterministic ordering
    for snapshot tests.
    """
    return tuple(sorted(_FLAGS.keys()))


def _utcnow_iso_z() -> str:
    """Return the current UTC time as an ISO-8601 string with a ``Z`` suffix.

    Centralised so every audit-log entry uses the same formatting
    (UTC, microsecond precision, ``Z`` instead of ``+00:00``).
    Python's ``datetime.isoformat()`` would emit ``+00:00`` by
    default, which is JSON-parseable but harder to grep and not
    the format every dashboard expects.
    """
    return (
        datetime.now(timezone.utc)
        .isoformat(timespec="microseconds")
        .replace("+00:00", "Z")
    )


def record_flag_change(
    flag_name: str,
    previous_value: bool,
    new_value: bool,
    *,
    no_op: bool = False,
) -> None:
    """Append one audit-log entry for a flag change and emit the log event.

    Single responsibility for the two coupled side effects of a
    flag mutation: (1) appending to the in-memory ``_HISTORY`` list
    so the GET ``/v1/feature_flags/history`` endpoint can serve it,
    and (2) emitting the ``feature_flag.changed`` structlog event so
    external log shippers can persist it. Doing both in one helper
    keeps the log-event shape and the stored-entry shape in sync:
    if a field is added to the log call, it must be added to the
    stored entry, and vice versa, and the call sites of this helper
    don't need to know about either.

    ``no_op=True`` marks a no-op write where ``previous_value`` ==
    ``new_value``. The audit log keeps these on purpose so the
    history captures every operator call (including duplicate
    toggles), but downstream consumers can filter on the field to
    isolate real state changes. ``no_op`` is keyword-only so a
    caller can't accidentally pass it positionally alongside the
    boolean values.

    The ``changed_at`` timestamp is captured here (not in the caller)
    so every entry is timestamped at the moment of recording, not
    at the moment of mutation — they are the same instant in
    practice, but this makes the helper independently testable
    without having to mock the caller's clock.
    """
    entry: dict[str, Any] = {
        "flag_name": flag_name,
        "previous_value": previous_value,
        "new_value": new_value,
        "changed_at": _utcnow_iso_z(),
        "no_op": no_op,
    }
    _HISTORY.append(entry)
    logger.info(
        "feature_flag.changed",
        flag_name=flag_name,
        previous_value=previous_value,
        new_value=new_value,
        no_op=no_op,
    )


def set_flag(name: str, enabled: bool) -> bool | None:
    """Set the on/off value of a registered feature flag.

    Looks up ``name`` in the in-process ``_FLAGS`` registry and,
    if present, replaces the value with ``enabled``. The previous
    value is returned so callers (notably the API toggle endpoint)
    can echo the change back to the operator without a second
    ``is_enabled()`` call.

    Unknown flag names return ``None`` (and emit a warning log
    event) rather than raising — the same deny-by-default contract
    that ``is_enabled()`` follows. A typo in the operator endpoint
    therefore fails closed: no registry mutation, no spurious
    "feature changed" event, and the caller can map the ``None``
    return to a 404.

    Pinned flags raise :class:`FlagPinnedError` rather than
    silently no-op'ing or returning a sentinel. The exception is
    distinct from the ``None`` return (unknown flag → 404) and
    from a successful mutation (returns ``previous_value``), so
    the API layer can map each outcome to the correct HTTP status
    without ambiguity. The mutation is rejected before the
    registry is touched, so a pinned flag cannot drift even if
    :func:`rollback_flag` is invoked — the rollback helper routes
    through :func:`set_flag` and therefore inherits the pin guard
    for free.

    A successful change (including no-op writes where ``enabled``
    equals the current value) appends an entry to the in-memory
    ``_HISTORY`` audit log via :func:`record_flag_change` and emits
    the matching ``feature_flag.changed`` info log event with the
    flag name, the previous value, the new value, and a ``no_op``
    flag. The audit log captures every operator call so dashboards
    and the ``GET /v1/feature_flags/history`` endpoint can
    attribute every runtime state change to a specific call without
    scraping structlog output.

    Design note (v17 hardening): the level is intentionally
    ``info``, not ``debug``. ``is_enabled()`` reserves ``warning``
    for the known-unknown path (typos / stale callers); the
    known-known mutation path stays at ``info`` regardless of
    whether the call mutated state. Distinguishing no-op vs. change
    at the log level would force dashboards to filter two event
    channels to answer "who flipped this flag" — the simpler,
    audit-friendly contract is one event, one level, with
    ``previous_value`` and ``new_value`` always present (and equal
    on no-op writes, so an operator can grep for ``previous_value ==
    new_value`` to isolate them).
    """
    if name not in _FLAGS:
        logger.warning(
            "feature_flag.unknown",
            flag_name=name,
            known_flags=sorted(_FLAGS.keys()),
        )
        return None
    if name in _PINNED_FLAGS:
        # Pin guard fires BEFORE any registry mutation or audit-log
        # append. A pinned flag cannot drift, and the rejection is
        # visible via the dedicated ``feature_flag.pinned`` warning
        # log event so operator dashboards can spot attempts to
        # toggle a locked flag without scraping structlog for the
        # post-hoc "did the value change?" check.
        logger.warning(
            "feature_flag.pinned",
            flag_name=name,
            current_value=_FLAGS[name],
        )
        raise FlagPinnedError(name, _FLAGS[name])
    previous_value = _FLAGS[name]
    new_value = bool(enabled)
    is_no_op = previous_value == new_value
    _FLAGS[name] = new_value
    record_flag_change(
        flag_name=name,
        previous_value=previous_value,
        new_value=new_value,
        no_op=is_no_op,
    )
    return previous_value


class FlagPinnedError(RuntimeError):
    """Raised when :func:`set_flag` is called on a pinned flag.

    Carries the flag name and its current value so the API layer
    can build a 423 Locked response with the exact flag id and the
    value the operator attempted to change. The exception is a
    ``RuntimeError`` subclass (not ``ValueError``) because the
    rejection is a runtime state condition, not a programming bug
    — the caller passed valid arguments and a known flag, but the
    flag is locked. Catching the specific subclass (rather than
    the bare ``RuntimeError``) lets the API layer isolate the
    423 path without affecting any other error handling.
    """

    def __init__(self, flag_name: str, current_value: bool) -> None:
        super().__init__(
            f"feature flag {flag_name!r} is pinned to {current_value}; "
            f"unpin it before mutating"
        )
        self.flag_name = flag_name
        self.current_value = current_value


def is_pinned(name: str) -> bool:
    """Return whether ``name`` is currently pinned.

    Unknown flags return ``False`` rather than raising — mirrors
    the deny-by-default contract of :func:`is_enabled`. The check
    is a single ``set`` membership lookup, so it is cheap to call
    on every :func:`set_flag` invocation.
    """
    return name in _PINNED_FLAGS


def pin_flag(name: str) -> dict[str, Any] | None:
    """Pin ``name`` so :func:`set_flag` rejects future mutations.

    The natural complement to v18's :func:`rollback_flag`: lets an
    operator mark a flag as locked without deleting it from the
    registry. Pin state is process-local and resets on every
    process restart — same lifecycle as ``_FLAGS`` and
    ``_HISTORY``.

    Returns a flat dict mirroring the ``FeatureFlagPinResponse``
    shape so the route can ``**``-unpack it without a
    field-by-field copy. Returns ``None`` for an unknown flag
    (route maps to 404). On a re-pin (already pinned) the helper
    is a no-op and returns the same shape with
    ``already_pinned=True`` so the caller can distinguish
    "pin succeeded" from "pin was already in effect" via the
    boolean rather than via an exception.

    The pin state itself is NOT recorded in ``_HISTORY`` — the
    audit log is for value mutations, not for meta-state. Pin
    events are visible only via the dedicated
    ``feature_flag.pinned`` (rejected mutation) and
    ``feature_flag.pinned_by_operator`` (successful pin) log
    events; if persistent audit is needed, subscribe a sink to
    those event names.
    """
    if name not in _FLAGS:
        logger.warning(
            "feature_flag.unknown",
            flag_name=name,
            known_flags=sorted(_FLAGS.keys()),
        )
        return None
    already_pinned = name in _PINNED_FLAGS
    if not already_pinned:
        _PINNED_FLAGS.add(name)
        logger.info(
            "feature_flag.pinned_by_operator",
            flag_name=name,
            current_value=_FLAGS[name],
        )
    return {
        "name": name,
        "pinned": True,
        "already_pinned": already_pinned,
        "current_value": _FLAGS[name],
    }


def unpin_flag(name: str) -> dict[str, Any] | None:
    """Remove ``name`` from the pinned set so future mutations succeed.

    Companion to :func:`pin_flag`. Returns ``None`` for an unknown
    flag (route maps to 404). On an unpin of an unpinned flag the
    helper is a no-op and returns the same shape with
    ``was_pinned=False`` so the route can render "no-op" without
    raising.
    """
    if name not in _FLAGS:
        logger.warning(
            "feature_flag.unknown",
            flag_name=name,
            known_flags=sorted(_FLAGS.keys()),
        )
        return None
    was_pinned = name in _PINNED_FLAGS
    if was_pinned:
        _PINNED_FLAGS.discard(name)
        logger.info(
            "feature_flag.unpinned_by_operator",
            flag_name=name,
            current_value=_FLAGS[name],
        )
    return {
        "name": name,
        "pinned": False,
        "was_pinned": was_pinned,
        "current_value": _FLAGS[name],
    }


def clear_pinned_flags() -> None:
    """Empty the in-memory pinned set.

    Intended for tests: every test that mutates ``_PINNED_FLAGS``
    should call this in a ``finally`` block (or via an autouse
    fixture) so subsequent tests start with no pins. Mirrors
    :func:`clear_flag_history`. Production code should NOT call
    this — pinning is the operator-facing "do not change this"
    signal, and clearing it programmatically would create a silent
    gap in the operator-visible intent.
    """
    _PINNED_FLAGS.clear()


def get_pinned_flags() -> tuple[str, ...]:
    """Return the sorted tuple of currently pinned flag names.

    Exposed for the operator endpoint and for tests that want to
    assert the pinned-set shape without re-parsing this module.
    The tuple is sorted so callers can rely on deterministic
    ordering for snapshot tests.
    """
    return tuple(sorted(_PINNED_FLAGS))


def get_flag_history() -> tuple[dict[str, Any], ...]:
    """Return the recorded audit log as a tuple, sorted ascending by ``changed_at``.

    The shape of each returned entry matches the
    ``FlagHistoryEntry`` Pydantic model that the
    ``GET /v1/feature_flags/history`` endpoint serialises, so the
    API layer can build the response without any field-by-field
    copy. Sorting is done here (not in the endpoint) so tests can
    rely on a stable order without re-sorting, and so the sort
    survives any future addition of a ring-buffer cap.

    The returned tuple is a fresh container — mutating it does
    not affect ``_HISTORY`` — but the dict elements inside it are
    shared references. Callers should treat the returned data as
    read-only; the helper does not deep-copy to keep the hot path
    cheap.
    """
    return tuple(sorted(_HISTORY, key=lambda entry: entry["changed_at"]))


def clear_flag_history() -> None:
    """Empty the in-memory audit log.

    Intended for tests: every test that mutates ``_FLAGS`` (and
    therefore records into ``_HISTORY``) should call this in a
    ``finally`` block (or via an autouse fixture) so subsequent
    tests start with a clean log. Mirrors the
    ``_KNOWN_PHASES_CACHE`` reset pattern used in
    ``generators/core/pack.py``.

    Production code should NOT call this — the audit log is the
    whole point of the feature, and a runtime call to
    ``clear_flag_history()`` would create a silent gap in the
    operator-visible trail. If a long-running process needs to
    bound memory, the right move is to add a ``maxlen`` ring buffer
    (see the ``_HISTORY`` docstring), not to clear on a schedule.
    """
    _HISTORY.clear()


# Maximum number of recent history rows the rollback helper scans
# before giving up. The audit log is bounded by design (each
# mutation is one append, and there are only three flags), so a
# cap of 100 rows is comfortably larger than any realistic
# operator-undo stack and small enough that the linear scan is
# trivially cheap. The cap is a safety net for the case where a
# noisy future client starts hammering the toggle endpoint — the
# cap ensures rollback stays O(1) in practice and never blocks the
# request thread on an unbounded list.
_ROLLBACK_SCAN_LIMIT: Final[int] = 100


def rollback_flag(name: str) -> dict[str, Any] | None:
    """Roll back the most recent real change to ``name``.

    Looks up the most recent non-no-op entry in ``_HISTORY`` for
    ``name`` (within the last ``_ROLLBACK_SCAN_LIMIT`` rows) and
    re-applies the entry's ``previous_value`` via :func:`set_flag`,
    effectively undoing the most recent mutation. The companion
    write is recorded in ``_HISTORY`` like every other flag
    mutation, so the rollback itself is visible to
    ``GET /v1/feature_flags/history`` as a normal ``no_op=False``
    entry.

    Return value is a flat dict matching the
    ``FeatureFlagRollbackResponse`` shape consumed by
    ``POST /v1/feature_flags/{name}/rollback``:

    - ``name`` — the flag that was rolled back
    - ``rolled_back_from`` — the value of the flag immediately
      before the rollback (``_FLAGS[name]`` at call time)
    - ``rolled_back_to`` — the value after the rollback, i.e.
      the ``previous_value`` of the most recent real change
    - ``restored_entry_index`` — the index (in the
      ``get_flag_history()`` sorted tuple) of the entry whose
      ``previous_value`` was re-applied. Useful for callers that
      want to render "rolled back change #N from the audit log".
      ``None`` if no real change exists (the function returned
      ``None`` — see below) so the route can mirror the same
      shape on the 409 conflict response.
    - ``history_size_at_rollback`` — the size of ``_HISTORY`` at
      the moment the rollback was recorded. Snapshot here (not in
      the route) so the value is consistent with the restored
      ``_FLAGS[name]`` regardless of any interleaved writes.

    The function returns ``None`` for both failure modes the
    caller needs to distinguish via HTTP status:

    - Unknown flag (``name not in _FLAGS``): mirrors
      :func:`set_flag`'s deny-by-default contract. ``set_flag``
      inside this helper will log ``feature_flag.unknown`` at
      warning level and return ``None`` itself, so the route can
      map this branch to a 404 without any extra detection.

    - No rollbackable history for ``name`` (every entry in the
      last 100 rows is either for a different flag or has
      ``no_op=True``, OR the flag is known but the audit log has
      no rows for it at all): the route maps this to a 409
      Conflict. The 409 is intentionally not a 404 — the flag
      exists in the registry, the operator's request was
      well-formed, but there is simply nothing to undo.

    No-op writes are intentionally ignored during the scan:
    ``no_op=True`` rows capture the operator's *call* but not a
    real state change, so restoring the row's ``previous_value``
    would re-apply a value the registry already holds. Skipping
    them keeps the rollback semantics aligned with "undo the last
    real mutation" rather than "undo the last call".

    The scan is reversed-then-iterated so we stop at the first
    match without scanning the entire history. The cap of
    ``_ROLLBACK_SCAN_LIMIT`` recent rows means the worst case is
    bounded even if the audit log grows large between flag
    changes.
    """
    # First gate: unknown flag → None (route maps to 404).
    if name not in _FLAGS:
        logger.warning(
            "feature_flag.unknown",
            flag_name=name,
            known_flags=sorted(_FLAGS.keys()),
        )
        return None

    # Reverse-scan the most recent ``_ROLLBACK_SCAN_LIMIT`` rows
    # for the first non-no-op change to ``name``. ``get_flag_history``
    # already returns rows sorted ascending by ``changed_at``, so
    # the most recent change is at the end; iterating from the end
    # hits the match in O(1) for the common single-roll case and
    # stays O(n) in the worst case, bounded by the scan-window cap.
    history = get_flag_history()
    target_entry: dict[str, Any] | None = None
    target_index: int = -1
    scan_window = history[-_ROLLBACK_SCAN_LIMIT:]
    for offset in range(1, len(scan_window) + 1):
        entry = scan_window[-offset]
        if entry["flag_name"] != name:
            continue
        if entry["no_op"]:
            continue
        target_entry = entry
        # Map the matched row back to its absolute index in the
        # full sorted history so callers can render "rolled back
        # change #N from the audit log".
        target_index = len(history) - offset
        break
    if target_entry is None:
        # No real change for this flag in the scan window (or
        # the audit log is empty for this flag). The route maps
        # this to 409 Conflict — the flag exists, the request was
        # well-formed, but there is nothing to undo.
        return None

    target_previous_value = target_entry["previous_value"]
    previous_value_before_rollback = set_flag(name, target_previous_value)

    # Defensive: ``set_flag`` only returns ``None`` for unknown
    # flags, which we already gated above, but a future refactor
    # could break that invariant — surface it as None rather than
    # silently lying about a successful rollback.
    if previous_value_before_rollback is None:
        return None

    return {
        "name": name,
        "rolled_back_from": previous_value_before_rollback,
        "rolled_back_to": target_previous_value,
        "restored_entry_index": target_index,
        "history_size_at_rollback": len(_HISTORY),
    }
