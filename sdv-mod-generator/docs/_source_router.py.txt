"""Intent routing: detect game → detect feature → return generator list.

Game detection happens first (keyword or LLM), then feature/phase routing
within the detected game pack.
"""
from typing import TypedDict

import structlog

# Bind a module-level logger so every structlog event carries the
# fully-qualified module name (``orchestrator.router``) by default.
# Other modules in this codebase do the same; the bare
# ``structlog.get_logger()`` form (no name argument) was a minor
# convention drift — addressed in v35 Blue for grep consistency.
logger = structlog.get_logger(__name__)


class RoutingHint(TypedDict):
    """Hint returned by router to orchestrator."""
    game: str
    phase: str
    generators: list[str]
    execution_order: list[str]
    dependencies: dict[str, list[str]]
    confidence: float  # 0.0-1.0 match confidence (1.0 = strong keyword, 0.0 = default fallback)
    matched_keyword: str  # the keyword that triggered the match (empty if fallback)

def detect_game(prompt: str) -> str:
    """Detect which game the prompt is about.

    Uses keyword matching first, falls back to LLM classification,
    defaults to 'stardew_valley'.
    """
    prompt_lower = prompt.lower()
    for game_id, keywords in _GAME_KEYWORDS.items():
        for kw in keywords:
            if kw in prompt_lower:
                logger.info("router.game_detected", game=game_id, keyword=kw)
                return game_id

    return "stardew_valley"


def route(prompt: str) -> tuple[str, RoutingHint]:
    """Match a prompt to game, phase, and generators.

    Two-stage routing:
    1. Detect game (keyword → LLM → stardew_valley fallback)
    2. Detect phase + generators within the game pack

    Returns:
        tuple of (phase, RoutingHint with game included)
    """
    game_id = detect_game(prompt)
    prompt_lower = prompt.lower()

    phase_map = _PHASE_BY_KEYWORD.get(game_id, {})
    matched_generators: list[str] = []
    matched_phase: str | None = None
    matched_keyword: str = ""
    best_keyword_len = 0

    for keyword, phase in phase_map.items():
        if keyword in prompt_lower and len(keyword) > best_keyword_len:
            best_keyword_len = len(keyword)
            matched_phase = phase
            matched_keyword = keyword

    # Weather-event priority override: when the prompt contains BOTH a
    # weather keyword (rain/storm/snow/wind/weather/buff) AND the generic
    # ``event`` word, prefer ``weather_event`` over ``event_mod``. Without
    # this, the longest-keyword-wins tie between ``"event"`` (5 chars) and
    # ``"storm"``/``"rain"`` (4-5 chars) resolves to ``event_mod`` purely
    # by dict-insertion order, even though semantically the user wants a
    # weather event (e.g. ``"add a rain storm event"``). Festival/event
    # prompts without any weather keyword still resolve to ``event_mod``
    # via the main loop above.
    if matched_phase == "event_mod" and any(
        k in prompt_lower for k in ("rain", "storm", "snow", "wind", "weather", "buff")
    ):
        matched_phase = "weather_event"
        matched_keyword = "event"  # longest keyword that drove the original match

    is_fallback = matched_phase is None
    if is_fallback:
        matched_phase = phase_map.get("shop", "shop_channel")
        matched_keyword = ""  # default fallback has no matched keyword

    # Confidence: heuristic based on the matched keyword's character length.
    # The longest real keyword in the maps is ~16 chars ("seasonal festival",
    # "map edit"). Matches of that length or longer get full confidence (1.0).
    # Single-word matches (3-4 chars) score low (0.2-0.25) so the orchestrator
    # can decide whether to ask the user for clarification. Fallback (no
    # keyword matched) is 0.0.
    if is_fallback or best_keyword_len == 0:
        confidence: float = 0.0
    else:
        confidence = min(1.0, round(best_keyword_len / 16.0, 2))

    try:
        from generators.core import get_game_pack
        pack = get_game_pack(game_id)
        if pack is None:
            logger.warning("router.pack_not_found", game=game_id, phase=matched_phase)
            matched_generators = _default_generators_for_phase(matched_phase)
        elif matched_phase not in pack.list_phases():
            logger.warning("router.phase_not_in_pack", game=game_id, phase=matched_phase)
            matched_generators = _default_generators_for_phase(matched_phase)
        else:
            pg = pack.get_generators(matched_phase)
            matched_generators = pg.execution_order.copy()
    except (ImportError, AttributeError, ValueError, TypeError) as exc:
        logger.warning("router.pack_fallback", game=game_id, phase=matched_phase, error=str(exc), error_type=type(exc).__name__)
        matched_generators = _default_generators_for_phase(matched_phase)

    hint: RoutingHint = {
        "game": game_id,
        "phase": matched_phase,
        "generators": matched_generators,
        "execution_order": matched_generators,
        "dependencies": {},
        "confidence": confidence,
        "matched_keyword": matched_keyword,
    }
    logger.info(
        "router.routed",
        game=game_id,
        phase=matched_phase,
        generators=matched_generators,
        confidence=confidence,
        matched_keyword=matched_keyword,
    )
    return matched_phase, hint


def _default_generators_for_phase(phase: str) -> list[str]:
    """Resolve a phase string to its defence-in-depth generator-name list.

    This is the fallback that :func:`route` consults when the upstream
    pack-based resolver (``StardewValleyPack.get_generators(phase)``) is
    unavailable or does not know the phase. Every phase advertised by
    ``StardewValleyPack.list_phases()`` must have a matching
    ``if phase == "..."`` arm here — the v100 hardening tests pin that
    bidirectional coverage at build time, and the v99 telemetry below
    pins the runtime signal for any drift that slips past CI.

    Args:
        phase: The phase identifier (e.g. ``"texture"``,
            ``"clothing_collection"``, ``"shop_channel"``) to resolve.
            Case-sensitive; matching is on the exact phase string.

    Returns:
        list[str]: The ordered list of generator class names that
        :func:`route` should run for ``phase``. The list is in execution
        order, never includes ``"manifest_generator"`` (the manifest is
        always the first generator emitted by the orchestrator, not part
        of the per-phase payload), and is non-empty for every phase
        declared in the pack's ``_MANIFEST.supported_phases``. An
        UNKNOWN phase (no matching ``if`` arm) returns ``[]`` and
        emits the ``router.default_generators.unknown`` WARNING so an
        operator can correlate a downstream "pipeline generated zero
        files" failure back to the specific phase string that fell
        through.

    Logging:
        * Matched phase: no log call here (the matched-phase success
          path is logged at the :func:`route` call site as
          ``router.default_generators.matched`` with the full routing
          context).
        * Unknown phase: emits ``router.default_generators.unknown``
          (WARNING, single ``phase`` snake_case field) immediately
          before returning ``[]``. WARNING (not INFO) because an
          unknown phase is always actionable — it indicates either a
          typo'd phase string or a pack drift (a phase added to the
          pack without the parallel fallback arm here).
    """
    # v99 Blue: the unknown-phase fallthrough at the bottom now
    # emits a canonical ``router.default_generators.unknown``
    # WARNING event before returning ``[]``. Previously this
    # fallthrough was silent — a typo'd phase, a phase removed
    # from the pack without updating this fallback, or a
    # future refactor that accidentally drops an ``if`` arm all
    # surfaced only as a downstream "pipeline generated zero
    # files" failure with no operator-actionable signal. The
    # function's contract is unchanged: returns ``list[str]``,
    # unknown phase -> empty list. This is purely additive
    # telemetry on the silent-failure path only; the matched-
    # phase success path is logged at the call site in
    # :func:`route` (``router.default_generators.matched``).
    if phase == "texture":
        return ["texture_generator"]
    if phase == "npc_schedule":
        return [
            "manifest_generator",
            "npc_schedule_generator",
            "npc_dialogue_generator",
            "npc_gift_taste_generator",
            "npc_content_json_generator",
        ]
    if phase == "shop_channel":
        return [
            "manifest_generator", "shop_item_pool_generator", "tv_channel_generator",
            "mail_system_generator", "item_sprites_generator", "ui_assets_generator",
            "catalog_preview_generator", "realism_damage_generator",
            "trigger_logic_generator", "config_schema_generator",
            "content_json_generator",
            # v53 Red: 12th shop_channel generator. Mirrors the
            # registration in
            # generators/packs/stardew_valley/__init__.py:236.
            "sound_effects_generator",
        ]
    if phase == "event_mod":
        return [
            "festival_schedule_generator",
            "festival_shop_generator",
            "festival_map_generator",
            "festival_dialogue_generator",
            "festival_mail_generator",
            "festival_content_json_generator",
        ]
    if phase == "custom_crafting":
        return [
            "crafting_recipe_generator",
            "cooking_recipe_generator",
            "crafting_content_json_generator",
        ]
    if phase == "farm_expansion":
        return [
            "manifest_generator",
            "building_generator",
            "warp_point_generator",
            "map_edit_generator",
            "farm_expansion_content_json_generator",
        ]

        # not pet gift preferences). The pack is the
        # first and only pack to edit ``Data/Pets``. The
        # defence-in-depth fallback excludes the
        # ``manifest_generator`` (the manifest is part of
        # the ``StardewValleyPack`` ``get_generators()``
        # wrapper, not the per-phase fallback) — mirrors
        # the v97 ``monster_slayer_quests``, v96b
        # ``fishponds``, v94 ``tool_definition``, and
        # earlier 2-gen / N-gen pack conventions.
        return [
            "pet_breed_definition_generator",
            "pet_breed_dialogue_generator",
            "pet_breed_gift_taste_generator",
            "pet_breed_sprite_generator",
            "pet_breed_content_json_generator",
        ]
    if phase == "horse_breed":
        # v99 Red: defence-in-depth fallback for the
        # 2-generator ``horse_breed`` shape (definition ->
        # content.json). The ``horse_breed`` pack edits
        # ``Data/Buildings`` (one additive ``Stable`` row per
        # custom breed token, with the per-row
        # ``MovementSpeed`` / ``CoatColor`` overrides) and
        # registers a small ``Strings/UI`` shim with the
        # per-breed display names. It is distinct from
        # ``pet_breed`` (v98, ``Data/Pets`` for cats/dogs —
        # different Content Patcher surface, different
        # lifecycle: pets are a follower, horses are a
        # mount), ``animal_expansion`` (``Data/Animals`` for
        # farm animals — cows, chickens, etc. — which live
        # in barns and coops), and ``furniture_definition``
        # (``Data/Furniture`` — horses are not placeable
        # furniture). The pack is the first and only pack
        # to add rows to the ``Stable`` token of
        # ``Data/Buildings``. The defence-in-depth fallback
        # excludes the ``manifest_generator`` (the manifest
        # is part of the ``StardewValleyPack``
        # ``get_generators()`` wrapper, not the per-phase
        # fallback) — mirrors the v98 ``pet_breed``, v97
        # ``monster_slayer_quests``, v96b ``fishponds``, v94
        # ``tool_definition``, and earlier 2-gen pack
        # conventions.
        return [
            "horse_breed_definition_generator",
            "horse_breed_content_json_generator",
        ]
    if phase == "clothing_collection":
        # v102 Red: defence-in-depth fallback for the
        # 2-generator ``clothing_collection`` shape
        # (definition -> content.json). The
        # ``clothing_collection`` pack edits
        # ``Data/Clothing`` (one additive
        # ``custom_shirt_<token>`` row per shirt) and
        # registers a small ``Strings/UI`` shim with the
        # per-shirt display name. It is distinct from
        # ``hat_collection`` (v90, ``Data/Hats`` — headwear
        # with ``ShowRealHair`` / ``Category`` envelope, no
        # ``Sleeves`` / ``OtherData`` fields),
        # ``boot_collection`` (v91, ``Data/Boots`` —
        # footwear with ``Defense`` / ``Immunity`` /
        # ``Color`` envelope, no ``Sleeves`` flag),
        # ``furniture_definition`` (v81, ``Data/Furniture``
        # — placeable items, not wearables),
        # ``object_definition`` (v78, ``Data/Objects`` —
        # pickup items, not wearables), ``npc_portrait``
        # (v82, ``Data/Characters`` portrait rows — not
        # clothing rows), and ``horse_breed`` (v87,
        # ``Data/Buildings`` ``Stable`` rows — not
        # clothing). The pack is the first and only pack
        # to edit ``Data/Clothing``. The defence-in-depth
        # fallback excludes the ``manifest_generator`` (the
        # manifest is part of the ``StardewValleyPack``
        # ``get_generators()`` wrapper, not the per-phase
        # fallback) — mirrors the v97 ``monster_slayer_quests``,
        # v96b ``fishponds``, v94 ``tool_definition``, v93
        # ``weapon_definition``, v92 ``custom_powers``, v91
        # ``boot_collection``, v90 ``hat_collection``, and
        # earlier 2-gen pack conventions.
        return [
            "clothing_collection_definition_generator",
            "clothing_collection_content_json_generator",
        ]
    # v99 Blue: unknown-phase silent-failure gap. The phase
    # did not match any of the 60+ ``if phase == "..."``
    # arms above, so we are about to return an empty list.
    # Emit a canonical ``router.default_generators.unknown``
    # WARNING so an operator can correlate a "pipeline
    # generated zero files" downstream failure back to the
    # specific phase string that fell through. The single
    # ``phase`` field is sufficient — the call site
    # (``route()``) already logs the full routing context
    # (``router.pack_not_found`` / ``router.phase_not_in_pack``
    # / ``router.pack_fallback``) so a log query can
    # pivot off this WARNING to surface the offending
    # phase. WARNING (not INFO) because an unknown phase
    # is always actionable — it indicates either a typo'd
    # phase string or a pack drift (a phase added to the
    # pack without the parallel fallback arm here).
    logger.warning("router.default_generators.unknown", phase=phase)
    return []
