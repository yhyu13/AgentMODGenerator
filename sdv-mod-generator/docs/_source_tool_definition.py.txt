"""Custom tool-definition feature pack for Stardew Valley.

Generates a Content Patcher mod that adds 1-3 custom
tools (Pickaxe / Axe / Hoe / Watering Can / Fishing
Rod variants) to the game by editing ``Data/Tools``
(one additive ``custom_tool_<token>`` row per tool)
and registering a small ``Strings/UI`` shim with the
per-tool display name + description.

Vanilla SDV's ``Data/Tools`` table is a flat dictionary
keyed by tool id (the vanilla keys are Title-Case
strings like ``"Pickaxe"``, ``"Axe"``, ``"Hoe"``,
``"Watering Can"``, ``"Pan"``); each row's body
carries a ``Name`` (lang key), ``Description`` (lang
key), ``AttachmentSlots`` (integer 0-4 — the number
of upgrade-forge slots the tool has), and a
``DisplayName`` lang key.

Distinct from neighbouring packs:

- ``weapon_definition`` (v93, ``Data/Weapons``) —
  weapons are equippable items with a damage envelope
  (``MinDamage`` / ``MaxDamage`` / ``CritChance`` /
  ``Speed``); tools are non-equippable utility items
  with no damage envelope and a forge-slots envelope
  (``AttachmentSlots``).
- ``boot_collection`` (v91, ``Data/Boots``) — boots
  are wearables with ``Defense`` / ``Immunity``
  envelopes; tools are not wearable.
- ``hat_collection`` (v90, ``Data/Hats``) — hats are
  headwear with no stat envelope; tools have a
  forge-slots envelope.
- ``object_definition`` (v78, ``Data/Objects``) —
  generic pickup items; tools are tools.
- ``furniture_definition`` (v81, ``Data/Furniture``) —
  placeable furniture; tools are not placeable.
- ``custom_powers`` (v92, ``Data/Powers``) — passive
  abilities shown in the Skills tab; tools are
  in-hand utility items.

The pack is composed of two cooperating generators
(definition -> content.json) that mirror the shape of
``npc_portrait`` (v82), ``npc_disposition`` (v80),
``furniture_definition`` (v81), ``horse_breed`` (v87),
``witch_swamp`` (v88), ``movie_theater`` (v89),
``hat_collection`` (v90), ``boot_collection`` (v91),
``custom_powers`` (v92), and ``weapon_definition``
(v93) — the smaller 2-gen packs:

1. :class:`ToolDefinitionDefinitionGenerator` —
   defines 1-3 custom tools. Each tool row carries
   an ``ItemId`` (snake_case, max 32 chars, must be
   prefixed with ``"custom_tool_"``), ``Name`` (short
   printable tool name, max 32 chars), ``Description``
   (one-line flavour text, max 128 chars),
   ``AttachmentSlots`` (int 0-4 — the vanilla SDV
   tool-upgrade slot count), ``Texture`` (asset path
   under ``Tools/``), and ``DisplayName`` (the lang
   key for the per-tool display name; we mirror
   ``"Tool.<ItemId>.Name"``). LLM-call + 3-tool
   hardcoded fallback (Bronze Pickaxe, Maple Hoe,
   Copper Axe) so the pipeline still produces a
   loadable mod when no LLM is configured.

2. :class:`ToolDefinitionContentJsonGenerator` —
   assembles the final ``content.json`` that edits
   ``Data/Tools`` (one additive ``custom_tool_<token>``
   row per tool) and registers a small ``Strings/UI``
   shim with the per-tool display name + description.
   Reads the manifest mod-id from ``prior_outputs``
   defensively with a hardcoded fallback.

This generator set is intentionally LLM-optional:
every generator has a deterministic fallback path so
the pipeline still produces a usable mod when no LLM
is configured. The fallback emits 3 custom tools
(``custom_tool_bronze_pickaxe``,
``custom_tool_maple_hoe``, ``custom_tool_copper_axe``)
that demonstrate the per-row additive pattern without
colliding with any vanilla SDV tool keys. The
``"custom_tool_"`` prefix keeps the additive
``Entries`` block safe.
"""
from __future__ import annotations

from pydantic import BaseModel, Field, ValidationError

import structlog

from generators.core import BaseGenerator, GeneratorInput, GeneratorOutput
from generators.llm_utils import generate_structured, llm_system_prompt
from generators.packs.stardew_valley._format_version import (
    FORMAT_VERSION as _FORMAT_VERSION,
)

logger = structlog.get_logger(__name__)


# Vanilla SDV tool ``Name`` lang-key template. The
# vanilla ``Data/Tools`` rows use the convention
# ``"Tool.<id>.Name"`` for the human-readable tool
# name and ``"Tool.<id>.Description"`` for the
# description text. We mirror that convention with
# the modded ``custom_tool_``-prefixed token.
_NAME_KEY_TEMPLATE: str = "Tool.{item_id}.Name"
_DESCRIPTION_KEY_TEMPLATE: str = "Tool.{item_id}.Description"


# Per-pack tool count envelope. 1-3 tools is the
# canonical sweet spot for an "add custom tools to my
# mod" prompt; cap at 3 to keep the per-row body of
# the content.json reasonable and to mirror the v82
# ``npc_portrait`` (3 NPCs) and v80 ``npc_disposition``
# (4 NPCs) envelopes. 3 is one smaller than v91
# ``boot_collection`` (4 boots) because tools are a
# more narrowly-scoped surface — players typically
# have 1-2 "primary" tool types per category.
_MIN_TOOLS: int = 1
_MAX_TOOLS: int = 3
_DEFAULT_TOOLS: int = 3


# Per-tool token length cap. Vanilla SDV tool ids
# are short, but Content Patcher can handle up to 64
# chars without issue; we cap at 32 to mirror the v93
# ``weapon_definition``, v92 ``custom_powers``, v91
# ``boot_collection``, and v90 ``hat_collection`` pack
# conventions.
_TOOL_TOKEN_MAX_LEN: int = 32


# Display name length cap. Vanilla SDV's tool-slot
# UI uses ~32 chars per tool display name. Mirrors
# the v93 ``weapon_definition`` and v92
# ``custom_powers`` pack conventions.
_DISPLAY_NAME_MAX_LEN: int = 32


# Tool description length cap. One-line flavour text
# shown beneath the tool name in the tool-slot UI;
# cap at 128 chars to allow a comfortable two-sentence
# description. Mirrors the v93 ``weapon_definition``
# pack convention.
_DESCRIPTION_MAX_LEN: int = 128


# Asset-path length cap. Vanilla SDV asset paths are
# short; cap at 96 chars to allow a reasonable
# ``Tools/<tool_token>``-style layout. Mirrors the
# v93 ``weapon_definition`` and v92 ``custom_powers``
# pack conventions.
_ASSET_PATH_MAX_LEN: int = 96


# Vanilla SDV ``Data/Tools`` attachment-slots
# envelope. The ``AttachmentSlots`` field is an
# integer 0-4 (vanilla SDV tools have 0-3 upgrade
# slots; a value of 4 is reserved for the most
# powerful mod-added tools). Out-of-range values
# would crash the tool-upgrade UI.
_MIN_ATTACHMENT_SLOTS: int = 0
_MAX_ATTACHMENT_SLOTS: int = 4
_DEFAULT_ATTACHMENT_SLOTS: int = 0


# Vanilla SDV uses Title-Case tool keys ("Pickaxe",
# "Axe", "Hoe", "Watering Can", "Pan"). The
# ``"custom_tool_"`` prefix keeps the additive
# ``Entries`` block from colliding with any vanilla
# SDV tool ids; we mirror the convention used by
# ``weapon_definition`` (v93), ``custom_powers``
# (v92), ``boot_collection`` (v91), ``hat_collection``
# (v90), ``furniture_definition`` (v81), and
# ``object_definition`` (v78).
_TOOL_TOKEN_PREFIX: str = "custom_tool"


# Default mod id fallback (matches the v93
# ``weapon_definition`` and v92 ``custom_powers`` pack
# convention of "lower-case mod id with a dot").
_DEFAULT_MOD_ID: str = "custom.tooldefinition"


# Default texture asset-path template. Used when the
# LLM does not supply a texture path so the
# deterministic fallback still produces a loadable
# Content Patcher manifest.
_DEFAULT_TEXTURE_TEMPLATE: str = "Tools/{tool_token}"


# ---------------------------------------------------------------------
# Sanitizers
# ---------------------------------------------------------------------


def _sanitize_tool_token(raw: object) -> str:
    """Return a safe snake_case tool token (max 32 chars).

    Mirrors :func:`_sanitize_weapon_token` from the v93
    ``weapon_definition`` pack and
    :func:`_sanitize_power_token` from the v92
    ``custom_powers`` pack: stripped to alphanumerics
    + underscore, lowercased, prefixed with
    ``"custom_tool_"`` if it would otherwise start
    with a digit or be missing the prefix, and capped
    at 32 chars. Always returns a non-empty string.
    """
    if raw is None:
        return f"{_TOOL_TOKEN_PREFIX}_default"
    text = str(raw)
    cleaned = "".join(
        c for c in text.lower() if c.isalnum() or c == "_"
    )
    if not cleaned:
        return f"{_TOOL_TOKEN_PREFIX}_default"
    if cleaned[0].isdigit():
        cleaned = f"{_TOOL_TOKEN_PREFIX}_" + cleaned
    if not cleaned.startswith(f"{_TOOL_TOKEN_PREFIX}_"):
        cleaned = f"{_TOOL_TOKEN_PREFIX}_" + cleaned
    return cleaned[:_TOOL_TOKEN_MAX_LEN] or (
        f"{_TOOL_TOKEN_PREFIX}_default"
    )


def _sanitize_attachment_slots(raw: object) -> int:
    """Coerce an attachment-slots value to an int in ``[0, 4]``.

    Mirrors :func:`_sanitize_damage` from the v93
    ``weapon_definition`` pack but specialised for
    the vanilla-SDV ``Data/Tools`` ``AttachmentSlots``
    envelope (int 0-4).
    """
    try:
        value = int(raw)  # type: ignore[arg-type]
    except (TypeError, ValueError):
        return _DEFAULT_ATTACHMENT_SLOTS
    if value < _MIN_ATTACHMENT_SLOTS:
        return _MIN_ATTACHMENT_SLOTS
    if value > _MAX_ATTACHMENT_SLOTS:
        return _MAX_ATTACHMENT_SLOTS
    return value


def _sanitize_display_name(raw: object) -> str:
    """Coerce a string into a printable display name (max 32 chars).

    Mirrors :func:`_sanitize_display_name` from the
    v93 ``weapon_definition`` and v92 ``custom_powers``
    packs: filters control characters, collapses
    whitespace, caps at 32 chars with an ellipsis
    suffix if truncated. Falls back to ``""`` for
    non-string inputs so the downstream content.json
    generator can swap in a default.
    """
    if not isinstance(raw, str):
        return ""
    cleaned = "".join(
        ch for ch in raw
        if ch.isprintable() or ch in ("\n", "\t", " ")
    )
    text = " ".join(cleaned.split()).strip()
    if len(text) > _DISPLAY_NAME_MAX_LEN:
        text = text[: _DISPLAY_NAME_MAX_LEN - 1].rstrip() + "\u2026"
    return text


def _sanitize_description(raw: object) -> str:
    """Coerce a string into a printable description (max 128 chars).

    Mirrors :func:`_sanitize_description` from the
    v93 ``weapon_definition`` and v92 ``custom_powers``
    packs: filters control characters, collapses
    whitespace, caps at 128 chars with an ellipsis
    suffix if truncated. Falls back to ``""`` for
    non-string inputs.
    """
    if not isinstance(raw, str):
        return ""
    cleaned = "".join(
        ch for ch in raw
        if ch.isprintable() or ch in ("\n", "\t", " ")
    )
    text = " ".join(cleaned.split()).strip()
    if len(text) > _DESCRIPTION_MAX_LEN:
        text = text[: _DESCRIPTION_MAX_LEN - 1].rstrip() + "\u2026"
    return text


def _sanitize_texture(
    raw: object, default_template: str, tool_token: str
) -> str:
    """Coerce an asset path string into a safe relative path.

    Falls back to the per-pack default template if the
    input is missing or invalid. Mirrors
    :func:`_sanitize_texture` from the v93
    ``weapon_definition`` and v92 ``custom_powers``
    packs.

    Hardening: any path-traversal segment (``..`` or
    absolute prefix) collapses to the default template
    so a malicious LLM response cannot smuggle
    ``../../../etc/passwd`` into the downstream
    ``content.json`` (which Content Patcher would
    then write to disk inside the generated mod
    zip).
    """
    if isinstance(raw, str) and raw.strip():
        text = raw.strip()
        if text.startswith("/"):
            text = text.lstrip("/")
        # Path-traversal guard — reject any segment
        # that is ``..`` or starts with ``..``
        # (handles both ``../foo`` and ``foo/../bar``
        # forms by splitting on forward slashes and
        # back-slashes). Content Patcher asset paths
        # use forward slashes only; we also strip
        # back-slashes to defend against
        # Windows-style traversal tokens.
        if ".." in text.replace("\\", "/").split("/"):
            return default_template.format(tool_token=tool_token)
        if text:
            return text[:_ASSET_PATH_MAX_LEN]
    return default_template.format(tool_token=tool_token)


def _sanitize_count(raw: object) -> int:
    """Coerce a value to an int clamped to ``[_MIN_TOOLS, _MAX_TOOLS]``.

    Falls back to ``_DEFAULT_TOOLS`` for non-numeric
    inputs. Mirrors :func:`_sanitize_count` from the
    v93 ``weapon_definition`` and v92 ``custom_powers``
    packs.
    """
    try:
        value = int(raw)  # type: ignore[arg-type]
    except (TypeError, ValueError):
        return _DEFAULT_TOOLS
    if value < _MIN_TOOLS:
        return _MIN_TOOLS
    if value > _MAX_TOOLS:
        return _MAX_TOOLS
    return value

class ToolSchema(BaseModel):
    """Schema for a single tool definition.

    Mirrors the per-row shape used by the v93
    ``weapon_definition`` pack's ``WeaponSchema`` but
    adapted for tool metadata. The 6 fields here
    cover the canonical SDV tool row: token id,
    display name, description, attachment-slot
    envelope, texture path, and display-name lang
    key.
    """

    ItemId: str = Field(
        ...,
        description=(
            "Snake_case tool id (prefixed custom_tool_, "
            "max 32 chars)"
        ),
    )
    Name: str = Field(..., description="Short printable tool name")
    Description: str = Field(
        ..., description="One-line flavour text (<= 128 chars)"
    )
    AttachmentSlots: int = Field(
        ..., description="Tool-upgrade slot count (int 0-4)"
    )
    Texture: str = Field(
        ..., description="Asset path under Tools/"
    )
    DisplayName: str = Field(
        ...,
        description=(
            "Lang key for the per-tool display name "
            "(typically 'Tool.<ItemId>.Name')"
        ),
    )


class ToolListSchema(BaseModel):
    """Wrapper for the LLM's structured-output response.

    The list is capped at ``_MAX_TOOLS * 2`` to
    defend against runaway LLM responses (mirrors the
    v93 ``weapon_definition`` and v92 ``custom_powers``
    pack conventions).
    """

    tools: list[ToolSchema] = Field(
        ...,
        max_length=_MAX_TOOLS * 2,
        description="Custom tools to add",
    )


# ---------------------------------------------------------------------
# Fallback data
# ---------------------------------------------------------------------


def _fallback_tool(
    *,
    item_id: str,
    name: str,
    description: str,
    attachment_slots: int,
) -> dict[str, object]:
    """Construct one curated tool row for the no-LLM path.

    Mirrors :func:`_fallback_weapon` from the v93
    ``weapon_definition`` pack but with the per-tool
    shape (6 fields vs 10 for weapon_definition —
    drops the damage / crit / speed / type envelope
    and replaces them with the simpler
    ``attachment_slots`` envelope).
    """
    texture_path = _DEFAULT_TEXTURE_TEMPLATE.format(
        tool_token=item_id
    )
    return {
        "ItemId": item_id,
        "Name": name,
        "Description": description,
        "AttachmentSlots": attachment_slots,
        "Texture": texture_path,
        "DisplayName": _NAME_KEY_TEMPLATE.format(
            item_id=item_id
        ),
    }


def _fallback_tool_list() -> list[dict[str, object]]:
    """Return the deterministic 3-tool fallback list.

    The 3 curated tools (Bronze Pickaxe, Maple Hoe,
    Copper Axe) cover three vanilla SDV tool
    archetypes (mining Pickaxe, farming Hoe, forestry
    Axe) so the fallback demonstrates the per-row
    additive pattern with realistic content. 3 tools
    is intentionally small (matches ``npc_portrait``'s
    fallback size) because the tool-slot UI is a
    tighter-scope surface with a more meaningful
    per-row design space — a 3-tool fallback is
    enough to demonstrate the pattern without
    padding with low-value entries.

    Mirrors :func:`_fallback_weapon_list` from the
    v93 ``weapon_definition`` pack and
    :func:`_fallback_hat_list` from the v90
    ``hat_collection`` pack.
    """
    return [
        _fallback_tool(
            item_id=f"{_TOOL_TOKEN_PREFIX}_bronze_pickaxe",
            name="Bronze Pickaxe",
            description=(
                "A sturdy bronze-headed pickaxe — breaks "
                "the toughest rocks in the mines."
            ),
            attachment_slots=2,
        ),
        _fallback_tool(
            item_id=f"{_TOOL_TOKEN_PREFIX}_maple_hoe",
            name="Maple Hoe",
            description=(
                "A lightweight hoe with a maple wood "
                "handle. Tills soil with ease."
            ),
            attachment_slots=1,
        ),
        _fallback_tool(
            item_id=f"{_TOOL_TOKEN_PREFIX}_copper_axe",
            name="Copper Axe",
            description=(
                "A sharp copper-bladed axe — fells trees "
                "in fewer swings than a starter axe."
            ),
            attachment_slots=2,
        ),
    ]

class ToolDefinitionDefinitionGenerator(BaseGenerator):
    """Define custom tools for a Stardew Valley mod."""

    name = "tool_definition_definition_generator"
    phase = "tool_definition"
    game = "stardew_valley"

    async def generate(
        self, inp: GeneratorInput
    ) -> GeneratorOutput:
        """Build the 1-3 tool definition list.

        On LLM failure, the deterministic fallback
        path emits 3 curated tools (Bronze Pickaxe,
        Maple Hoe, Copper Axe) so the pipeline still
        produces a loadable Content Patcher manifest.
        Mirrors the v93 ``weapon_definition`` and v92
        ``custom_powers`` packs'
        :meth:`DefinitionGenerator.generate`.
        """
        out = GeneratorOutput()
        prompt = (
            f"Create custom Data/Tools rows for a "
            f"Stardew Valley mod based on: "
            f"\"{inp['prompt']}\"\n\n"
            "Generate 1-3 unique tool entries. For each "
            "tool provide:\n"
            "- ItemId: snake_case id, max 32 chars, "
            "must be prefixed with \"custom_tool_\"\n"
            "- Name: short printable tool name, max 32 chars\n"
            "- Description: one-line flavour text, max "
            "128 chars\n"
            "- AttachmentSlots: int between 0 and 4 "
            "(vanilla SDV tool-upgrade slot count)\n"
            "- Texture: relative asset path under "
            "Tools/<tool_token>/\n"
            "- DisplayName: lang key for the per-tool "
            "display name (typically 'Tool.<ItemId>.Name')\n\n"
            "Respond with ONLY valid JSON matching the "
            "expected schema."
        )

        try:
            result = await generate_structured(
                prompt,
                ToolListSchema,
                system=llm_system_prompt(),
                max_tokens=4096,
            )
            parsed = ToolListSchema(**result)
            tools: list[dict[str, object]] = []
            for t in parsed.tools[:_MAX_TOOLS]:
                tools.append(
                    {
                        "ItemId": t.ItemId,
                        "Name": t.Name,
                        "Description": t.Description,
                        "AttachmentSlots": t.AttachmentSlots,
                        "Texture": t.Texture,
                        "DisplayName": t.DisplayName,
                    }
                )
            if not tools:
                raise ValueError("no tools produced")
        except (
            ValueError, RuntimeError, IOError, ValidationError
        ) as exc:
            logger.error(
                "tool_definition_definition_generator.failed",
                error=str(exc),
                error_type=type(exc).__name__,
            )
            tools = _fallback_tool_list()

        # Sanitize each row defensively. A malicious
        # or malformed LLM response cannot bypass
        # these clamps.
        sanitized: list[dict[str, object]] = []
        for row in tools:
            sanitized.append(_sanitize_tool_row(row))

        out.add_file(
            "assets/tool_definition/tools.json",
            {"tools": sanitized},
        )
        out.metadata["tool_count"] = len(sanitized)
        out.metadata["tool_ids"] = [
            t["ItemId"] for t in sanitized
        ]
        return out

    def validate_output(self, output: GeneratorOutput) -> list[str]:
        """Pin the per-row shape contract.

        Required keys, tool list non-empty, each row
        has ``ItemId`` + ``Name`` + ``Texture`` +
        ``AttachmentSlots``, ``ItemId`` starts with
        the ``"custom_tool_"`` prefix,
        ``AttachmentSlots`` is in the vanilla-SDV
        ``[0, 4]`` range.
        """
        errors: list[str] = []
        data = output.files.get(
            "assets/tool_definition/tools.json"
        )
        if not data:
            errors.append(
                "tool_definition_definition_generator: "
                "assets/tool_definition/tools.json missing"
            )
            return errors
        if not isinstance(data, dict):
            errors.append(
                "tool_definition_definition_generator: "
                "tools.json must be a dict"
            )
            return errors
        tools = data.get("tools")
        if (
            not isinstance(tools, list)
            or len(tools) < _MIN_TOOLS
        ):
            errors.append(
                "tool_definition_definition_generator: "
                "tools list missing or too short"
            )
            return errors
        seen_tokens: set[str] = set()
        for t in tools:
            if not isinstance(t, dict):
                errors.append(
                    "tool_definition_definition_generator: "
                    "each tool item must be a dict"
                )
                continue
            for required in (
                "ItemId",
                "Name",
                "Texture",
                "AttachmentSlots",
                "Description",
                "DisplayName",
            ):
                if required not in t:
                    errors.append(
                        "tool_definition_definition_generator: "
                        f"missing {required!r}"
                    )
            token = t.get("ItemId")
            if isinstance(token, str):
                if token.lower() in seen_tokens:
                    errors.append(
                        "tool_definition_definition_generator: "
                        f"duplicate ItemId {token!r}"
                    )
                seen_tokens.add(token.lower())
                if not token.startswith(f"{_TOOL_TOKEN_PREFIX}_"):
                    errors.append(
                        "tool_definition_definition_generator: "
                        f"ItemId {token!r} missing "
                        f"'{_TOOL_TOKEN_PREFIX}_' prefix"
                    )
            slots = t.get("AttachmentSlots")
            if (
                not isinstance(slots, int)
                or slots < _MIN_ATTACHMENT_SLOTS
                or slots > _MAX_ATTACHMENT_SLOTS
            ):
                errors.append(
                    "tool_definition_definition_generator: "
                    f"AttachmentSlots {slots!r} out of "
                    f"range [{_MIN_ATTACHMENT_SLOTS}, "
                    f"{_MAX_ATTACHMENT_SLOTS}]"
                )
        return errors


class ToolDefinitionContentJsonGenerator(BaseGenerator):
    """Assemble the final content.json for Content Patcher."""

    name = "tool_definition_content_json_generator"
    phase = "tool_definition"
    game = "stardew_valley"

    async def generate(
        self, inp: GeneratorInput
    ) -> GeneratorOutput:
        """Build the content.json by emitting one
        Data/Tools EditData change (per tool) plus a
        small Strings/UI shim for the display names +
        descriptions.

        Reads the definition generator's prior output
        and the manifest mod-id defensively with a
        hardcoded fallback. Mirrors the v93
        ``weapon_definition`` pack's
        :class:`ContentJsonGenerator`.
        """
        out = GeneratorOutput()
        prior = inp.get("prior_outputs", {}) or {}

        # Manifest mod id (lower-case per Content
        # Patcher convention). Falls back to the
        # hard-coded mod id if the manifest generator
        # didn't run first.
        manifest_data = prior.get(
            "manifest_generator", GeneratorOutput()
        )
        mod_id = _DEFAULT_MOD_ID
        if isinstance(manifest_data, GeneratorOutput):
            manifest = manifest_data.files.get("manifest.json")
            if isinstance(manifest, dict):
                unique = manifest.get("UniqueID", "")
                if isinstance(unique, str) and unique.strip():
                    mod_id = unique.strip().lower()

        # Pull tool list from the definition generator.
        tools: list[dict[str, object]] = []
        definition_data = prior.get(
            "tool_definition_definition_generator",
            GeneratorOutput(),
        )
        if isinstance(definition_data, GeneratorOutput):
            tools_file = definition_data.files.get(
                "assets/tool_definition/tools.json"
            )
            if isinstance(tools_file, dict):
                raw_list = tools_file.get("tools", [])
                if isinstance(raw_list, list):
                    tools = [
                        t for t in raw_list
                        if isinstance(t, dict)
                    ]

        # Defensive fallback: no valid prior outputs
        # -> emit the deterministic 3-tool fallback
        # list so the content.json still loads.
        if not tools:
            tools = _fallback_tool_list()

        # Build per-tool Data/Tools rows and the
        # per-tool Strings/UI shim entries.
        tool_entries: dict[str, dict[str, object]] = {}
        strings_entries: dict[str, str] = {}
        for row in tools:
            token = row.get("ItemId", "")
            if not isinstance(token, str) or not token:
                continue
            display = _sanitize_display_name(
                row.get("Name", "")
            )
            if not display:
                # Fallback display name from token so
                # the downstream content.json doesn't
                # crash on an empty string. Strip the
                # ``custom_tool_`` prefix and title-case
                # the remainder.
                bare = token[
                    len(_TOOL_TOKEN_PREFIX) + 1:
                ]
                display = (
                    bare.replace("_", " ").title()
                    or token
                )
            description = _sanitize_description(
                row.get("Description", "")
            )
            texture = _sanitize_texture(
                row.get("Texture", ""),
                _DEFAULT_TEXTURE_TEMPLATE,
                token,
            )
            attachment_slots = _sanitize_attachment_slots(
                row.get(
                    "AttachmentSlots",
                    _DEFAULT_ATTACHMENT_SLOTS,
                )
            )
            display_name = row.get("DisplayName", "")
            if (
                not isinstance(display_name, str)
                or not display_name.strip()
            ):
                display_name = (
                    _NAME_KEY_TEMPLATE.format(item_id=token)
                )
            description_key = _DESCRIPTION_KEY_TEMPLATE.format(
                item_id=token
            )

            tool_entries[token] = {
                "Name": display_name,
                "DisplayName": display_name,
                "Description": description_key,
                "AttachmentSlots": attachment_slots,
                "Texture": texture,
            }
            strings_entries[display_name] = display
            strings_entries[description_key] = description

        content: dict[str, object] = {
            "Format": _FORMAT_VERSION,
            "Changes": [
                {
                    "Action": "EditData",
                    "Target": "Data/Tools",
                    "Entries": tool_entries,
                },
                {
                    "Action": "EditData",
                    "Target": "Strings/UI",
                    "Entries": strings_entries,
                },
            ],
        }

        out.add_file("content.json", content)
        out.metadata["mod_id"] = mod_id
        out.metadata["tool_count"] = len(tools)
        out.metadata["tool_entry_count"] = len(tool_entries)
        return out

    def validate_output(self, output: GeneratorOutput) -> list[str]:
        """Pin the content.json shape contract.

        content.json is a dict, has ``Format`` and
        ``Changes``, ``Changes[0]["Target"]`` is
        ``"Data/Tools"``, every ``ItemId`` starts
        with ``"custom_tool_"``, every tool row
        carries the canonical ``Name`` / ``Texture``
        / ``AttachmentSlots`` fields, and the
        ``Strings/UI`` change is present.
        """
        errors: list[str] = []
        content = output.files.get("content.json")
        if not content:
            errors.append(
                "tool_definition_content_json_generator: "
                "content.json missing"
            )
            return errors
        if not isinstance(content, dict):
            errors.append(
                "tool_definition_content_json_generator: "
                "content.json must be a dict"
            )
            return errors
        if "Format" not in content:
            errors.append(
                "tool_definition_content_json_generator: "
                "content.json missing Format field"
            )
        if "Changes" not in content:
            errors.append(
                "tool_definition_content_json_generator: "
                "content.json missing Changes field"
            )
        changes = content["Changes"]
        if not isinstance(changes, list) or not changes:
            errors.append(
                "tool_definition_content_json_generator: "
                "content.json Changes must be a non-empty list"
            )
            return errors

        targets = [
            c.get("Target", "")
            for c in changes
            if isinstance(c, dict)
        ]
        if "Data/Tools" not in targets:
            errors.append(
                "tool_definition_content_json_generator: "
                "content.json missing Data/Tools change"
            )
        if "Strings/UI" not in targets:
            errors.append(
                "tool_definition_content_json_generator: "
                "content.json missing Strings/UI change"
            )

        tools_change = next(
            (
                c for c in changes
                if isinstance(c, dict)
                and c.get("Target") == "Data/Tools"
            ),
            None,
        )
        if tools_change is not None:
            entries = tools_change.get("Entries", {})
            if not isinstance(entries, dict) or not entries:
                errors.append(
                    "tool_definition_content_json_generator: "
                    "Data/Tools Entries must be a non-empty dict"
                )
            else:
                for key, row in entries.items():
                    if not isinstance(row, dict):
                        errors.append(
                            "tool_definition_content_json_generator: "
                            f"Data/Tools Entries[{key}] not a dict"
                        )
                        continue
                    for required in (
                        "Name",
                        "Texture",
                        "AttachmentSlots",
                        "DisplayName",
                        "Description",
                    ):
                        if required not in row:
                            errors.append(
                                "tool_definition_content_json_generator: "
                                f"Data/Tools Entries[{key}] "
                                f"missing {required!r}"
                            )
                    if not key.startswith(f"{_TOOL_TOKEN_PREFIX}_"):
                        errors.append(
                            "tool_definition_content_json_generator: "
                            f"Data/Tools Entries[{key}] key "
                            f"missing '{_TOOL_TOKEN_PREFIX}_' prefix"
                        )
        return errors

def _sanitize_tool_row(row: dict[str, object]) -> dict[str, object]:
    """Apply per-field sanitizers to one LLM-emitted row.

    Mirrors the in-place sanitization done by the v93
    ``weapon_definition`` pack's
    :func:`_sanitize_weapon_row`, factored into a
    helper for readability given the per-tool shape
    (6 fields vs 10 for weapon_definition).
    """
    raw_token = row.get("ItemId", "")
    token = _sanitize_tool_token(raw_token)
    name = _sanitize_display_name(row.get("Name", ""))
    if not name:
        bare = token[len(_TOOL_TOKEN_PREFIX) + 1:]
        name = bare.replace("_", " ").title() or token
    description = _sanitize_description(
        row.get("Description", "")
    )
    texture = _sanitize_texture(
        row.get("Texture", ""),
        _DEFAULT_TEXTURE_TEMPLATE,
        token,
    )
    attachment_slots = _sanitize_attachment_slots(
        row.get("AttachmentSlots", _DEFAULT_ATTACHMENT_SLOTS)
    )
    display_name = row.get("DisplayName", "")
    if (
        not isinstance(display_name, str)
        or not display_name.strip()
    ):
        display_name = _NAME_KEY_TEMPLATE.format(
            item_id=token
        )
    return {
        "ItemId": token,
        "Name": name,
        "Description": description,
        "AttachmentSlots": attachment_slots,
        "Texture": texture,
        "DisplayName": display_name,
    }
